\section{Rangkuman}
\begin{itemize}
    \item Persoalan \textit{knapsack} mencari bit $b_i \in \{0,1\}$ dengan
          $\sum_i b_i w_i = M$. Versi umumnya tergolong \textit{NP-complete},
          sehingga diyakini sukar dipecahkan dalam waktu polinomial
          (Persamaan~\eqref{eq:knapsack-dasar}).
    \item Barisan \textit{superincreasing} memenuhi
          $w_i > \sum_{j<i} w_j$. Setiap elemen lebih besar daripada jumlah
          semua elemen sebelumnya (Persamaan~\eqref{eq:superincreasing}).
    \item Solusi \textit{superincreasing knapsack} dapat ditemukan secara
          \textit{greedy} dalam waktu linear $O(n)$: bila $M \ge w_n$ maka
          $b_n = 1$, lalu prosedur diulang pada sisa bobot.
    \item Kemudahan itu menjadi kelemahan. Siapa pun yang mengetahui barisannya
          dapat mendekripsi pesan, sehingga barisan \textit{superincreasing}
          tidak boleh dipakai langsung sebagai kunci.
    \item Merkle--Hellman menyembunyikan barisan \textit{superincreasing}
          $W_{priv}$ menjadi kunci publik
          $w'_i = (w_i \cdot n) \bmod m$. Syaratnya $m > \sum_i w_i$ dan
          $\text{PBB}(n,m)=1$ (Persamaan~\eqref{eq:mh-kunci}).
    \item Syarat $m > \sum_i w_i$ mencegah pemutaran modulo saat dekripsi.
          Syarat $\text{PBB}(n,m)=1$ menjamin $n$ mempunyai invers modulo $m$.
    \item Enkripsi menjumlahkan elemen publik yang bitnya $1$,
          $C = \sum_i b_i w'_i$. Penjumlahan itu tidak diambil modulo $m$,
          sehingga $C$ dapat melampaui $m$ (Persamaan~\eqref{eq:mh-enkripsi}).
    \item Dekripsi menghitung $C' = (C \cdot n^{-1}) \bmod m$
          (Persamaan~\eqref{eq:mh-dekripsi}), lalu memecahkan $C'$ dengan
          algoritma \textit{greedy} atas $W_{priv}$.
    \item Kebenaran dekripsi bersandar pada
          $C' \equiv \sum_i b_i w_i \pmod m$, karena
          $n \cdot n^{-1} \equiv 1$ (Persamaan~\eqref{eq:mh-bukti}). Karena
          $0 \le \sum_i b_i w_i \le \sum_i w_i < m$, kesamaan modulo itu menjadi
          kesamaan biasa (Persamaan~\eqref{eq:mh-bukti-eksak}).
    \item Contoh lengkap: dengan $W_{priv} = \{2,3,6,13,27,52\}$, $m = 105$,
          dan $n = 31$ diperoleh $W_{pub} = \{62,93,81,88,102,37\}$ dan
          $n^{-1} = 61$. Pesan \code{011000110101101110} menjadi
          $(174, 280, 333)$ dan pulih tepat seperti semula.
    \item Ukuran cipherteks lebih besar daripada plainteks. Sumber
          menganjurkan paling sedikit 250 elemen berukuran 200 sampai 400 bit,
          tetapi modulus 100 sampai 200 bit tidak konsisten dengan syarat
          $m > \sum_i w_i$. Untuk 250 elemen sekecil itu, modulus perlu paling
          sedikit 449 bit.
    \item Keamanan Merkle--Hellman patah. Pada 1982 Shamir merekonstruksi
          barisan \textit{superincreasing} dari kunci publik dalam waktu
          polinomial (Bagian~\ref{sec:kriptanalisis-knapsack}).
\end{itemize}
