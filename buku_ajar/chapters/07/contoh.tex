\section{Contoh Terhitung}

\begin{example}
\label{ex:des-sbox}
\textbf{Membaca kotak-S DES.} Kotak-S menerima 6 bit. Bit pertama dan keenam
menentukan nomor baris, bit kedua sampai kelima menentukan nomor kolom. Pakai
tabel $S_1$ pada Tabel~\ref{tab:sbox-des}.

\begin{center}
\small
\begin{tabular}{@{}lllll@{}}
\hline
\textbf{Masukan} & \textbf{Baris} & \textbf{Kolom} & \textbf{Entri} & \textbf{Keluaran} \\
\hline
\code{011000} & $00_2=0$ & $1100_2=12$ & 5 & \code{0101} \\
\code{110100} & $10_2=2$ & $1010_2=10$ & 9 & \code{1001} \\
\code{011011} & $01_2=1$ & $1101_2=13$ & 5 & \code{0101} \\
% Koreksi: slide halaman 23 menulis luaran 13 untuk masukan 101010.
% Dengan tabel S1 baku, baris 2 kolom 5 bernilai 6.
\code{101010} & $10_2=2$ & $0101_2=5$ & 6 & \code{0110} \\
\hline
\end{tabular}
\end{center}

Baris ke-2 tabel $S_1$ (dihitung dari baris 0, lihat
Tabel~\ref{tab:sbox-des}) berbunyi
\begin{center}
\small
\setlength{\tabcolsep}{4pt}
\begin{tabular}{@{}*{16}{c}@{}}
4 & 1 & 14 & 8 & 13 & 6 & 2 & 11 & 15 & 12 & 9 & 7 & 3 & 10 & 5 & 0
\end{tabular}
\end{center}
sehingga entri pada kolom ke-10 bernilai $9$. Keluaran 4 bit inilah yang
kemudian dipermutasi oleh $P$.
\end{example}

\begin{example}
\label{ex:des-subkunci}
\textbf{Pembangkitan subkunci pertama DES.} Pakai kunci
$K = \code{133457799BBCDFF1}$ dan tabel PC-1 serta PC-2 pada
Tabel~\ref{tab:pc-des}.

Kunci itu dalam biner, byte demi byte, adalah
\begin{center}
\texttt{0001001\underline{1}\ 0011010\underline{0}\ 0101011\underline{1}\ 0111100\underline{1}}\\
\texttt{1001101\underline{1}\ 1011110\underline{0}\ 1101111\underline{1}\ 1111000\underline{1}}
\end{center}
dengan bit terakhir setiap byte (bit pada posisi $8, 16, \dots, 64$)
digarisbawahi. Kedelapan bit itu adalah bit paritas yang dibuang oleh PC-1.

Membaca PC-1 baris demi baris pada 64 bit itu, empat baris pertama menghasilkan
\[
C_0 = 1111\,0000\,1100\,1100\,1010\,1010\,1111 = \code{F0CCAAF},
\]
dan empat baris terakhir menghasilkan
\[
D_0 = 0101\,0101\,0110\,0110\,0111\,1000\,1111 = \code{556678F}.
\]
Perhatikan bahwa tidak satu pun bilangan $8, 16, 24, 32, 40, 48, 56, 64$ muncul
pada PC-1, sehingga kedelapan bit paritas itu benar-benar terbuang.

Putaran pertama memakai pergeseran $s_1 = 1$ bit, jadi
\[
C_1 = \text{geser}(C_0) = \code{E19955F}, \qquad
D_1 = \text{geser}(D_0) = \code{AACCF1E}.
\]
Membaca PC-2 pada $C_1 \parallel D_1$ menghasilkan subkunci pertama
\[
K_1 = 000110\,110000\,001011\,101111\,111111\,000111\,000001\,110010 = \code{1B02EFFC7072},
\]
yaitu 48 bit seperti yang disyaratkan fungsi $f$. Subkunci kedua, dengan
pergeseran $s_2 = 1$ bit lagi, adalah
\[
K_2 = \code{79AED9DBC9E5}.
\]
Perhatikan bahwa $K_1$ dan $K_2$ berbeda; itulah tanda bahwa jadwal pergeseran
dan PC-2 bekerja sebagaimana mestinya.
% Seluruh nilai di atas diverifikasi ulang dengan Python (des_uji.py).
\end{example}

\begin{example}
\label{ex:des-putaran-pertama}
\textbf{Putaran pertama DES lengkap.} Contoh ini mengenkripsi sebuah plainteks
dengan kunci tertentu dan menghasilkan vektor uji resmi berikut.
\begin{center}
\begin{tabular}{@{}ll@{}}
Plainteks  & $P = \code{0123456789ABCDEF}$ \\
Kunci      & $K = \code{133457799BBCDFF1}$ \\
Cipherteks & $C = \code{85E813540F0AB405}$ \\
\end{tabular}
\end{center}
Seluruh langkahnya dapat dihitung dari tabel pada Tabel~\ref{tab:ip-des},
Tabel~\ref{tab:ep-des}, Tabel~\ref{tab:sbox-des}, dan
Tabel~\ref{tab:pc-des}.

\textbf{Langkah 1 --- permutasi awal.} Membaca IP pada 64 bit plainteks
menghasilkan dua separuh
\[
L_0 = \code{CC00CCFF}, \qquad R_0 = \code{F0AAF0AA}.
\]

\textbf{Langkah 2 --- ekspansi dan XOR subkunci.} Dari
Contoh~\ref{ex:des-subkunci} diperoleh $K_1=\code{1B02EFFC7072}$. Membaca tabel
$E$ pada $R_0=\code{F0AAF0AA}$ yang dalam biner adalah
$1111\,0000\,1010\,1010\,1111\,0000\,1010\,1010$ menghasilkan
\[
E(R_0) = 011110\,100001\,010101\,010101\,011110\,100001\,010101\,010101 = \code{7A15557A1555},
\]
dan sesudah di-XOR dengan $K_1$,
\[
\begin{aligned}
E(R_0) \oplus K_1 &= 011000\,010001\,011110\,111010\,100001\,100110\,010100\,100111 \\
&= \code{6117BA866527}.
\end{aligned}
\]

\textbf{Langkah 3 --- substitusi kotak-S.} Kedelapan kelompok enam bit itu
disubstitusi satu per satu memakai tabel $S_1$ sampai $S_8$:

\begin{center}
\small
\begin{tabular}{@{}cccccc@{}}
\hline
$j$ & \textbf{Masukan} & \textbf{Baris} & \textbf{Kolom} & \textbf{$S_j$} & \textbf{Keluaran} \\
\hline
1 & \code{011000} & 0 & 12 & 5  & \code{0101} \\
2 & \code{010001} & 1 & 8  & 12 & \code{1100} \\
3 & \code{011110} & 0 & 15 & 8  & \code{1000} \\
4 & \code{111010} & 2 & 13 & 2  & \code{0010} \\
5 & \code{100001} & 3 & 0  & 11 & \code{1011} \\
6 & \code{100110} & 2 & 3  & 5  & \code{0101} \\
7 & \code{010100} & 0 & 10 & 9  & \code{1001} \\
8 & \code{100111} & 3 & 3  & 7  & \code{0111} \\
\hline
\end{tabular}
\end{center}

Hasil penggabungan kedelapan keluaran itu adalah
\[
S\text{-out} = \code{5C82B597}.
\]

\textbf{Langkah 4 --- permutasi $P$.} Membaca tabel $P$ pada $\code{5C82B597}$
menghasilkan
\[
P(S) = \code{234AA9BB}.
\]

\textbf{Langkah 5 --- pembentukan separuh baru.} Subkunci $K_1$ tidak lagi
dipakai; yang tersisa hanyalah XOR dengan separuh kiri:
\[
R_1 = L_0 \oplus P(S) = \code{CC00CCFF} \oplus \code{234AA9BB} = \code{EF4A6544},
\qquad L_1 = R_0 = \code{F0AAF0AA}.
\]

Kelima belas putaran berikutnya berjalan dengan pola yang sama, memakai
$K_2, K_3, \dots, K_{16}$. Hasil antara pada setiap putaran tercantum pada
tabel berikut.

\begin{center}
\small
\setlength{\tabcolsep}{5pt}
\begin{tabular}{@{}ccc@{}}
\hline
$i$ & $L_i$ & $R_i$ \\
\hline
0 & \code{CC00CCFF} & \code{F0AAF0AA} \\
1 & \code{F0AAF0AA} & \code{EF4A6544} \\
2 & \code{EF4A6544} & \code{CC017709} \\
3 & \code{CC017709} & \code{A25C0BF4} \\
4 & \code{A25C0BF4} & \code{77220045} \\
5 & \code{77220045} & \code{8A4FA637} \\
6 & \code{8A4FA637} & \code{E967CD69} \\
7 & \code{E967CD69} & \code{064ABA10} \\
8 & \code{064ABA10} & \code{D5694B90} \\
\end{tabular}
\hspace{1.6em}
\begin{tabular}{@{}ccc@{}}
\hline
$i$ & $L_i$ & $R_i$ \\
\hline
9  & \code{D5694B90} & \code{247CC67A} \\
10 & \code{247CC67A} & \code{B7D5D7B2} \\
11 & \code{B7D5D7B2} & \code{C5783C78} \\
12 & \code{C5783C78} & \code{75BD1858} \\
13 & \code{75BD1858} & \code{18C3155A} \\
14 & \code{18C3155A} & \code{C28C960D} \\
15 & \code{C28C960D} & \code{43423234} \\
16 & \code{43423234} & \code{0A4CD995} \\
\end{tabular}
\end{center}

Karena jaringan Feistel tidak menukar kedua separuh pada putaran terakhir,
blok sebelum permutasi akhir adalah $R_{16} \parallel L_{16}$, yaitu
\code{0A4CD99543423234}. Membaca $\text{IP}^{-1}$ pada blok itu memberikan
cipherteks
\[
C = \code{85E813540F0AB405}.
\]
Untuk dekripsi, jalankan 16 putaran yang sama dengan urutan subkunci dibalik,
yaitu $K_{16}$ lebih dahulu dan $K_1$ terakhir.

% termasuk S-out = 5C82B597 (bukan 5C82B5A5).
\end{example}

\begin{example}
\label{ex:des-kunci-lemah}
\textbf{Mengapa kunci} \code{0101010101010101} \textbf{lemah.} Kunci itu dalam
biner adalah delapan byte identik
\begin{center}
\texttt{00000001\ 00000001\ 00000001\ 00000001}\\
\texttt{00000001\ 00000001\ 00000001\ 00000001}
\end{center}
dan bit pada posisi $8, 16, \dots, 64$ adalah bit terakhir setiap byte.
Karena PC-1 membuang tepat kedelapan posisi itu, tujuh bit yang tersisa pada
setiap byte semuanya bernilai $0$. Jadi
\[
C_0 = 0000\,0000\,0000\,0000\,0000\,0000\,0000, \quad
D_0 = 0000\,0000\,0000\,0000\,0000\,0000\,0000.
\]
Pergeseran siklik atas deretan nol tetap menghasilkan deretan nol, sehingga
$C_i = D_i = 0$ untuk semua $i$ dan PC-2 menghasilkan
$K_i = \code{000000000000}$ untuk setiap putaran. Karena semua subkunci sama,
fungsi $f$ yang dipakai pada setiap putaran identik. Akibatnya
$E_K(E_K(x)) = x$ untuk setiap blok $x$. Inilah yang membuat kunci itu
dinyatakan lemah.
% Subkunci nol dan sifat E_K(E_K(x)) = x diverifikasi dengan Python.
\end{example}
