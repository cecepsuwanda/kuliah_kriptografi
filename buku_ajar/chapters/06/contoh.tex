\section{Contoh Terhitung}

\begin{example}
\textbf{ECB dan CBC pada plainteks yang sama.} Cipher blok mainan berukuran
4 bit memakai $E_K(P) = (P \oplus K) \ll 1$ dan
$D_K(C) = (C \gg 1) \oplus K$. Plainteksnya \code{1010 0010 0011 1010 1001}
(heksa \code{A23A9}), kuncinya $K = \code{1011}$ (heksa \code{B}), dan
$IV = \code{0000}$. Blok ke-1 dan ke-4 sama, yaitu \code{1010}.

\textbf{Mode ECB.} Setiap blok dienkripsi sendiri-sendiri:
\begin{enumerate}
    \item $E_K(\code{1010}) = \code{0010}$.
    \item $E_K(\code{0010}) = \code{0011}$.
    \item $E_K(\code{0011}) = \code{0001}$.
    \item $E_K(\code{1010}) = \code{0010}$.
    \item $E_K(\code{1001}) = \code{0100}$.
\end{enumerate}
Cipherteksnya \code{0010 0011 0001 0010 0100}, yaitu \code{23124} dalam heksa.
Blok \code{1010} yang berulang menghasilkan cipherteks \code{0010} yang sama,
sehingga pola plainteks tetap tampak.

\textbf{Mode CBC.} Setiap blok di-XOR dulu dengan cipherteks sebelumnya:
\begin{enumerate}
    \item $C_1 = E_K(\code{1010} \oplus \code{0000}) = \code{0010}$.
    \item $C_2 = E_K(\code{0010} \oplus \code{0010}) = \code{0111}$.
    \item $C_3 = E_K(\code{0011} \oplus \code{0111}) = \code{1111}$.
    \item $C_4 = E_K(\code{1010} \oplus \code{1111}) = \code{1101}$.
    \item $C_5 = E_K(\code{1001} \oplus \code{1101}) = \code{1111}$.
\end{enumerate}
% Koreksi: slide menulis 27FBF. Hitung ulang memberi 0010 0111 1111 1101 1111
% = 27FDF, dan dekripsi mengembalikan plainteks semula.
Cipherteksnya \code{0010 0111 1111 1101 1111}, yaitu \code{27FDF} dalam heksa.
Blok \code{1010} di posisi 1 dan 4 kini menjadi \code{0010} dan \code{1101},
sehingga pengulangan tersembunyi.

\textbf{Dekripsi CBC.} Dengan $P_i = D_K(C_i) \oplus C_{i-1}$, blok pertama
menjadi $D_K(\code{0010}) \oplus \code{0000} = \code{1010} \oplus \code{0000} =
\code{1010}$. Blok berikutnya kembali menjadi \code{0010}, \code{0011},
\code{1010}, dan \code{1001}. Jadi seluruh plainteks \code{A23A9} diperoleh
kembali.
\end{example}
