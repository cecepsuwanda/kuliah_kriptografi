\section{Rangkuman}
\begin{itemize}
    \item Fungsi hash adalah jenis algoritma kriptografi yang ketiga, setelah
          algoritma simetri dan algoritma nir-simetri.
    \item Fungsi hash mengompresi pesan berukuran sembarang menjadi
          \textit{message-digest} berukuran tetap.
    \item Fungsi hash bersifat \textit{irreversible}: pesan asli tidak dapat
          diperoleh kembali dari nilai hash.
    \item Kegunaan fungsi hash adalah memeriksa integritas pesan.
    \item Contoh algoritma fungsi hash adalah MD5, SHA-1, SHA-2, Keccak (SHA-3),
          RIPEMD, dan WHIRLPOOL.
    \item \textit{Message Authentication Code} (MAC) tidak dibahas dalam modul
          sumber.
\end{itemize}
