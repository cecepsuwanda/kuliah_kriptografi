\section{Contoh Terhitung}

\begin{example}
\textbf{Caesar cipher dengan $k=3$.} Plainteks \code{temui saya di jembatan
merah nanti malam} dienkripsi dengan pergeseran tiga huruf. Huruf \code{t}
$=19$ menjadi $(19+3) \bmod 26=22=\code{W}$. Huruf \code{e} $=4$ menjadi
$7=\code{H}$. Huruf \code{m} $=12$ menjadi $15=\code{P}$. Huruf \code{u} $=20$
menjadi $23=\code{X}$. Huruf \code{i} $=8$ menjadi $11=\code{L}$. Cipherteks
lengkapnya adalah \code{WHPXL VDBD GL MHPEDWDQ PHUDK QDQWL PDODP}.
\end{example}

\begin{example}
\textbf{Vigen\`ere cipher dengan kunci \code{soreang}.} Plainteks yang dipakai
adalah
\begin{center}
\code{terimapesanangulaikambing}.
\end{center}
Panjangnya 25 huruf. Kunci \code{soreang} bernilai $(18,14,17,4,0,13,6)$ dan
diulang menjadi \code{soreangsoreangsoreangsore}. Tiga huruf pertama dihitung
sebagai $(19+18) \bmod 26=11=\code{L}$, $(4+14) \bmod 26=18=\code{S}$, dan
$(17+17) \bmod 26=8=\code{I}$. Cipherteks lengkapnya adalah
\begin{center}
\code{LSIMMNVWGRRAAMMZRMKNSTWEK}.
\end{center}
\end{example}

\begin{example}
\textbf{One-Time Pad.} Plainteks \code{onetimepad} dan kunci
\code{tbfrgfarhm} berpanjang sama, yaitu sepuluh huruf. Enkripsi menghasilkan
\code{HOJKOREGHP}. Huruf pertama adalah
$(\code{o}+\code{t})=(14+19) \bmod 26=7=\code{H}$. Dekripsi huruf pertama
adalah $(\code{H}-\code{t})=(7-19) \bmod 26=14=\code{o}$.

\end{example}
