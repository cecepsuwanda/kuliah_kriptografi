
\begin{obereflection}
\textbf{Latihan 12.1}
\begin{enumerate}
    \item Jelaskan persoalan logaritma diskret. Mengapa keamanan algoritma
          ElGamal bergantung pada persoalan itu?
    \item Jelaskan definisi akar primitif dari bilangan prima $p$. Bandingkan
          $a = 3$ dan $a = 2$ untuk $p = 7$.
    \item Diketahui $p = 2357$, $g = 2$, dan kunci privat $x = 1751$. Hitunglah
          kunci publik $y$, lalu enkripsikan $m = 2035$ dengan $k = 1520$.
          Periksa bahwa dekripsinya memang mengembalikan $2035$.
    \item Diketahui $7$ adalah akar primitif dari $41$ dan $7^{3} = 343 \equiv 15
          \pmod{41}$. Jelaskan mengapa $x = 3$ merupakan logaritma diskret dari
          $15$ untuk basis $7$ modulo $41$.
    \item Mengapa invers $\left(a^{x}\right)^{-1}$ dapat dihitung sebagai
          $a^{\,p-1-x} \bmod p$? Tunjukkan hasil kalinya dengan $a^{x}$.
    \item Mengapa setiap blok pesan harus berada di dalam selang $[0, p-1]$?
    \item Pada contoh ``HALO'' dengan $p = 2273$, mengapa $m = 07001114$ tidak
          dapat dienkripsi sebagai satu blok? Jelaskan cara memecahnya.
    \item Pada contoh yang sama, nilai $0700$ dihitung sebagai $700$. Mengapa
          hal itu tidak mengubah hasil dekripsi?
\end{enumerate}
\end{obereflection}

\begin{obereflection}
\textbf{Latihan 12.2}
\begin{enumerate}
    \item Alice memilih $p = 2273$, $g = 3$, dan $x = 243$. Hitunglah
          $y = g^{x} \bmod p$ dan bandingkan hasilnya dengan $461$.
    \item Enkripsikan blok $m_{1} = 0700$ dengan $k = 1463$ dan blok
          $m_{2} = 1114$ dengan $k = 2001$ memakai kunci publik
          $(461, 3, 2273)$. Periksa bahwa cipherteksnya adalah $(1439, 74)$ dan
          $(1220, 1682)$.
    \item Dekripsikan kedua cipherteks pada soal 2 dengan kunci privat
          $x = 243$. Tunjukkan bahwa hasilnya kembali menjadi ``HALO''.
    \item Apa yang ditambahkan ElGamal terhadap skema distribusi kunci
          Diffie--Hellman?
    \item Bandingkan landasan keamanan RSA dan ElGamal. Persoalan apa yang
          menjadi dasar masing-masing sistem?
    \item Pada Contoh~\ref{ex:elgamal-halo}, kedua blok memakai nilai $k$ yang
          berbeda. Komponen $a = g^{k} \bmod p$ tidak bergantung pada pesan
          $m$. Jelaskan mengapa demikian.
    \item Sebutkan besaran yang boleh disiarkan dan besaran yang harus
          dirahasiakan pada algoritma ElGamal.
\end{enumerate}
\end{obereflection}
