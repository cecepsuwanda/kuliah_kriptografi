\section{Rangkuman}
\begin{itemize}
    \item Algoritma ElGamal diperkenalkan oleh Taher Elgamal pada tahun 1985.
    Keamanannya terletak pada sulitnya menghitung logaritma diskret.
    \item Makalah ElGamal memuat penerapan skema distribusi kunci
    Diffie--Hellman sebagai sistem kriptografi kunci-publik dan sebuah skema
    tanda tangan.
    \item Akar primitif $a$ modulo bilangan prima $p$ adalah bilangan yang
    perpangkatannya $a^{1}, a^{2}, \dots, a^{p-1}$ menghasilkan seluruh residu
    bukan-nol modulo $p$ tepat satu kali masing-masing.
    \item Untuk $p = 7$, $a = 3$ adalah akar primitif karena menghasilkan
    $(3, 2, 6, 4, 5, 1)$, sedangkan $a = 2$ bukan akar primitif karena
    menghasilkan $(2, 4, 1)$ yang berulang.
    \item Logaritma diskret dari $b$ untuk basis $a$ modulo $p$ adalah pangkat
    $x$ yang memenuhi $b \equiv a^{x} \pmod p$. Persoalan logaritma diskret
    adalah mencari $x$ bila $b$, $a$, dan $p$ diketahui.
    \item Pembangkitan kunci: pilih prima $p$ dan akar primitif $g$, pilih
    kunci privat $x$ dengan $2 \le x \le p-2$, lalu hitung kunci publik
    $y = g^{x} \bmod p$ (Persamaan~\eqref{eq:elgamal-kunci}).
    \item Enkripsi: pilih bilangan acak $k$ dengan $1 \le k \le p-2$, hitung
    $a = g^{k} \bmod p$ dan $b = y^{k} \cdot m \bmod p$, sehingga cipherteksnya
    berupa pasangan $(a, b)$ (Persamaan~\eqref{eq:elgamal-a}
    dan~\eqref{eq:elgamal-b}). Ukuran cipherteks dua kali ukuran plainteks.
    \item Dekripsi: hitung $\left(a^{x}\right)^{-1} \equiv a^{\,p-1-x}
    \pmod p$, lalu $m = b \cdot \left(a^{x}\right)^{-1} \bmod p$
    (Persamaan~\eqref{eq:elgamal-invers} dan~\eqref{eq:elgamal-dekripsi}).
    \item Kebenaran dekripsi bersandar pada identitas $\frac{b}{a^{x}} \equiv
    \frac{g^{xk} m}{g^{xk}} \equiv m \pmod p$, yang meniadakan faktor $g^{xk}$
    pada pembilang dan penyebut (Persamaan~\eqref{eq:elgamal-bukti}).
    \item Contoh pertama: $p = 2357$, $g = 2$, dan $x = 1751$ memberi
    $y = 1185$; pesan $m = 2035$ dengan $k = 1520$ menjadi $(a, b) = (1430,
    697)$ dan kembali menjadi $2035$ setelah dekripsi.
    \item Contoh kedua: $p = 2273$, $g = 3$, dan $x = 243$ memberi $y = 461$.
    Kata ``HALO'' dengan pengodean $A = 00$ menjadi \code{07001114}, terpecah
    menjadi $m_{1} = 0700$ dan $m_{2} = 1114$, yang menghasilkan cipherteks
    $(1439, 74)$ dan $(1220, 1682)$ dengan $k = 1463$ dan $k = 2001$.
\end{itemize}
