\textbf{Kuis Bab 12}
\begin{enumerate}
    \item Jelaskan persoalan logaritma diskret dan hubungannya dengan keamanan
          algoritma ElGamal.
    \item Jelaskan langkah-langkah pembangkitan kunci pada sistem ElGamal.
    \item Diketahui $p = 2273$, $g = 3$, dan $x = 243$. Hitunglah $y$, lalu
          enkripsikan $m = 700$ dengan $k = 1463$ dan periksa hasil dekripsinya.
    \item Tunjukkan bahwa $g = 2$ bukan akar primitif dari $p = 7$, sedangkan
          $g = 3$ adalah akar primitifnya.
    \item Jelaskan mengapa $\left(a^{x}\right)^{-1}$ dapat dihitung sebagai
          $a^{\,p-1-x} \bmod p$.
    \item Tunjukkan bahwa $m$ dapat dipulihkan dari pasangan cipherteks
          $(a, b)$.
\end{enumerate}
\end{obeassessment}
