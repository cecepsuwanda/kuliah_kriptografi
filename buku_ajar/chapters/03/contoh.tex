\section{Contoh Terhitung}

\begin{example}[Aritmetika modulo dan invers]
Hitung beberapa nilai berikut.
\begin{enumerate}
    \item $-24 \bmod 11$. Karena $-24 = (-3) \cdot 11 + 9$, hasilnya $9$.
    \item $4^{-1} \bmod 9$. Karena $4 \cdot 7 = 28 = 3 \cdot 9 + 1$, inversnya
          $7$.
    \item $23^{-1} \bmod 10$. Karena $23 \cdot (-3) = -69 = (-7) \cdot 10 + 1$,
          inversnya $-3 \equiv 7 \pmod{10}$.
\end{enumerate}
\end{example}

\begin{example}[Entropi Shannon sebuah distribusi]
Pesan $X = \code{AABBCBDB}$ terdiri atas 8 simbol dengan peluang $p(A)=2/8$,
$p(B)=4/8$, $p(C)=1/8$, dan $p(D)=1/8$. Substitusi ke rumus entropi memberi
\[
\begin{aligned}
H(X) &= -\sum_{i} p(x_i) \log_2 p(x_i)\\
     &= -\left(\tfrac14 \log_2 \tfrac14 + \tfrac12 \log_2 \tfrac12
        + \tfrac18 \log_2 \tfrac18 + \tfrac18 \log_2 \tfrac18\right)\\
     &= -\left(\tfrac14(-2) + \tfrac12(-1) + \tfrac18(-3) + \tfrac18(-3)\right)\\
     &= 1{,}75\ \text{bit/simbol}.
\end{aligned}
\]
Entropi maksimum untuk empat simbol adalah $\log_2 4 = 2$ bit. Distribusi yang
tidak seragam menghasilkan entropi yang lebih rendah.
\end{example}

\begin{example}[Entropi plainteks berita Kota Ternate]
Plainteks berita Kota Ternate pada Bagian~\ref{sec:teori-informasi} memuat 296
huruf setelah spasi dan tanda baca dibuang. Sebaran hurufnya tidak rata: huruf A
muncul 59 kali, N 34 kali, dan E 29 kali. Substitusi ke rumus entropi memberi
\[ H(P) = 3{,}8988\ \text{bit/karakter}, \]
atau sekitar $82{,}95\%$ dari entropi maksimum 26 huruf. Setelah dienkripsi,
entropi cipherteks menjadi
\[ H(C) = 4{,}3035\ \text{bit/karakter}, \]
atau sekitar $91{,}55\%$ dari maksimum. Entropi cipherteks lebih tinggi karena
sebaran hurufnya lebih rata.
\end{example}
