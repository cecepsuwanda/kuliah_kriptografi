\section{Contoh Terhitung}

% Sumber: hlm. 5, 8--10, 16--19, 25--27.

Seluruh bilangan heksadesimal pada bagian ini ditulis tanpa awalan \code{0x}.
Setiap contoh dihitung ulang dengan program agar angka pada tabel dan narasi
sama.

\begin{example}[Pembentukan kunci putaran RC5-32/0/1]
\label{ex:rc5-jadwal}
Tinjau RC5 dengan $w = 32$ bit, $r = 0$ putaran, dan kunci eksternal satu byte
$K = \code{00}$. Karena $r = 0$, tidak ada putaran. Contoh ini memperlihatkan
seluruh proses pembangkitan kunci putaran.

\textbf{Langkah 1: menghitung jumlah kunci.} $t = 2r + 2 = 2$, yaitu $S[0]$ dan
$S[1]$. Karena $b = 1$ byte dan satu kata berukuran 4 byte,
$c = \lceil 1/4 \rceil = 1$. Larik $L$ beranggota satu kata, yaitu
$L[0] = \code{00000000}$. Banyaknya langkah pencampuran adalah
$n = 3 \cdot \max(t,c) = 3 \cdot 2 = 6$.

\textbf{Langkah 2: inisialisasi larik $S$.} Dari
Tabel~\ref{tab:konstanta-pw-qw}, $P_{32} = \code{B7E15163}$ dan
$Q_{32} = \code{9E3779B9}$. Maka
\[
S[0] = \code{B7E15163}, \qquad
S[1] = \code{B7E15163} + \code{9E3779B9} = \code{5618CB1C}.
\]

\textbf{Langkah 3: pencampuran pertama.} Mula-mula $i = j = X = Y = 0$. Langkah
pertama mengerjakan
\[
S[0] \leftarrow \mathrm{ROTL}_{32}(\code{B7E15163} + 0 + 0,\ 3) = \code{BF0A8B1D}.
\]
Nilai $X$ menjadi \code{BF0A8B1D}, dan $i$ maju menjadi 1. Langkah $L$ memakai
$X$ yang baru dan $Y$ yang masih nol:
\[
L[0] \leftarrow \mathrm{ROTL}_{32}(\code{00000000} + \code{BF0A8B1D} + 0,\ \code{BF0A8B1D}).
\]
Besar pergeseran adalah \code{BF0A8B1D} modulo 32, yaitu \code{1D} $= 29$. Jadi
\[
L[0] = \mathrm{ROTL}_{32}(\code{BF0A8B1D}, 29) = \code{B7E15163}, \qquad
Y \leftarrow \code{B7E15163}.
\]

\textbf{Langkah 4: lima langkah berikutnya.} Aturannya sama. Karena $c = 1$,
indeks $j$ selalu nol. Hasil kelima langkah itu ditabulasikan pada
Tabel~\ref{tab:jejak-jadwal-kunci-rc5}.

\textbf{Hasil.} Kunci putarannya adalah $S[0] = \code{4DBA7B7A}$ dan
$S[1] = \code{1E1D1179}$. Karena $r = 0$, enkripsi hanya menjumlahkan kedua kata
plainteks dengan $S[0]$ dan $S[1]$. Plainteks nol karena itu menghasilkan
cipherteks $(S[0], S[1])$.
\end{example}

\begin{example}[Satu enkripsi RC5 dengan $w = 16$]
\label{ex:rc5-kecil}
Tinjau RC5 dengan $w = 16$ bit (blok 32 bit) dan $r = 2$ putaran. Blok plainteks
dipecah menjadi $A_0 = \code{1234}$ dan $B_0 = \code{2345}$. Kunci putarannya
\[
\begin{aligned}
S[0] &= \code{1A2B}, & S[1] &= \code{0C0D}, & S[2] &= \code{3344},\\
S[3] &= \code{5566}, & S[4] &= \code{7788}, & S[5] &= \code{1122}.
\end{aligned}
\]
Keenam nilai $S[i]$ dipilih agar perhitungannya pendek. Seluruh penjumlahan
dilakukan modulo $2^{16}$.

\textbf{Langkah 1: penyaringan awal.}
\[ A = \code{1234} + \code{1A2B} = \code{2C5F} = 11359, \]
\[ B = \code{2345} + \code{0C0D} = \code{2F52} = 12114. \]

\textbf{Langkah 2: putaran $i = 1$, bagian $A$.}
\[ A \leftarrow \mathrm{ROTL}_{16}(A \oplus B,\; B) + S[2]. \]
\[ A \oplus B = \code{2C5F} \oplus \code{2F52} = \code{030D} = 781,\qquad B \bmod 16 = 2, \]
\[ \mathrm{ROTL}_{16}(\code{030D}, 2) = \code{0C34} = 3124,\qquad A = 3124 + 13124 = \code{3F78}. \]

\textbf{Langkah 3: putaran $i = 1$, bagian $B$.}
\[ B \leftarrow \mathrm{ROTL}_{16}(B \oplus A,\; A) + S[3]. \]
\[ B \oplus A = \code{2F52} \oplus \code{3F78} = \code{102A} = 4138,\qquad A \bmod 16 = 8, \]
\[ \mathrm{ROTL}_{16}(\code{102A}, 8) = \code{2A10} = 10768,\qquad B = 10768 + 21862 = \code{7F76}. \]

\textbf{Langkah 4: putaran $i = 2$, bagian $A$.}
\[ A \leftarrow \mathrm{ROTL}_{16}(\code{3F78} \oplus \code{7F76},\; \code{7F76}) + S[4], \]
\[ \code{3F78} \oplus \code{7F76} = \code{400E} = 16398,\qquad \mathrm{ROTL}_{16}(\code{400E}, 6) = \code{0390} = 912, \]
\[ A = 912 + 30600 = \code{7B18} = 31512. \]

\textbf{Langkah 5: putaran $i = 2$, bagian $B$.}
\[ B \leftarrow \mathrm{ROTL}_{16}(\code{7F76} \oplus \code{7B18},\; \code{7B18}) + S[5], \]
\[ \code{7F76} \oplus \code{7B18} = \code{046E} = 1134,\qquad \mathrm{ROTL}_{16}(\code{046E}, 8) = \code{6E04} = 28164, \]
\[ B = 28164 + 4386 = \code{7F26} = 32550. \]

\textbf{Hasil.} Cipherteksnya adalah $(A,B) = (\code{7B18},\code{7F26})$.
\end{example}

\begin{example}[Satu putaran GOST]
\label{ex:gost-round}
Fungsi $f$ GOST menjumlahkan $R_{i-1}$ dengan kunci, mensubstitusi hasilnya
dengan delapan kotak-S, lalu menggesernya 11 bit. Tinjau keadaan awal putaran
ke-$i$ dengan $L_{i-1} = \code{01020304}$ dan $R_{i-1} = \code{12345678}$.
Kunci putarannya $K_i = \code{61626364}$, yaitu empat byte pertama dari kunci
contoh pada Subbab~\ref{sec:gost}.

\textbf{Langkah 1: penjumlahan modulo $2^{32}$.}
\[
R_{i-1} + K_i = \code{12345678} + \code{61626364} = \code{7396B9DC}.
\]

\textbf{Langkah 2: substitusi kedelapan kotak-S.} Hasil penjumlahan dipecah
menjadi delapan nibble, dimulai dari nibble paling kanan. Nibble \code{C} masuk
$S_1$, nibble \code{D} masuk $S_2$, dan seterusnya sampai nibble \code{7} masuk
$S_8$. Pembacaan Tabel~\ref{tab:kotak-s-gost} menghasilkan nilai berikut.

\begin{table}[htbp]
    \centering
    \small
    \begin{tabular}{@{}l *{8}{c}@{}}
    \toprule
    \textbf{Nibble ke-} & 1 & 2 & 3 & 4 & 5 & 6 & 7 & 8 \\
    \midrule
    Masukan              & \code{C} & \code{D} & \code{9} & \code{B} & \code{6} & \code{9} & \code{3} & \code{7} \\
    Kotak-S              & $S_1$ & $S_2$ & $S_3$ & $S_4$ & $S_5$ & $S_6$ & $S_7$ & $S_8$ \\
    Luaran               & \code{7} & \code{7} & \code{F} & \code{C} & \code{D} & \code{6} & \code{1} & \code{4} \\
    \bottomrule
    \end{tabular}
    \caption{Substitusi kedelapan nibble pada fungsi $f$ GOST untuk masukan
    \code{7396B9DC}}
    \label{tab:jejak-substitusi-gost}
    \sumber{Dihitung ulang dari kotak-S pada hlm.~8.}
\end{table}

Sebagai pemeriksaan, $S_1$ berisi permutasi $4, 10, 9, 2, 13, 8, 0, 14, 6, 11, 1,
12, 7, 15, 5, 3$. Masukan \code{C} sama dengan 12, sehingga elemen ke-12
dihitung dari nol adalah 7. Untuk $S_6$, masukan \code{9} menghasilkan 6.

\textbf{Langkah 3: penggabungan luaran.} Kedelapan luaran digabung menjadi satu
bilangan 32 bit. Nibble dari $S_1$ menempati posisi paling kanan:
\[
S_1 \| S_2 \| \cdots \| S_8 = \code{4}\,\code{1}\,\code{6}\,\code{D}\,\code{C}\,\code{F}\,\code{7}\,\code{7}
= \code{416DCF77}.
\]

\textbf{Langkah 4: pergeseran sirkuler.}
\[
\mathrm{ROTL}_{32}(\code{416DCF77}, 11) = \code{6E7BBA0B}.
\]

\textbf{Langkah 5: XOR dengan setengah blok kiri.} Menurut
Persamaan~\eqref{eq:putaran-gost},
\[
R_i = L_{i-1} \oplus f(R_{i-1}, K_i) = \code{01020304} \oplus \code{6E7BBA0B} = \code{6F79B90F}.
\]
Setengah blok kiri yang baru adalah setengah blok kanan yang lama, yaitu
$L_i = \code{12345678}$. Keadaan pada akhir putaran adalah
$(L_i, R_i) = (\code{12345678}, \code{6F79B90F})$.

\textbf{Hasil.} Contoh ini berhenti pada satu putaran. Cipherteks lengkap
diperoleh setelah 32 putaran dengan kunci menurut
Tabel~\ref{tab:jadwal-kunci-gost}.
\end{example}
