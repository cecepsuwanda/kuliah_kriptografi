
\begin{obeassessment}
\textbf{Kuis Bab 2}
\begin{enumerate}
    \item Apa perbedaan mendasar antara serangan pasif dan serangan aktif? Manakah yang
          lebih sulit dideteksi? Jelaskan alasannya.
    \item Dalam kondisi apa sebuah algoritma yang secara matematis kuat tetap bisa
          ditembus menggunakan \textit{Side-Channel Attack}?
    \item Diketahui cipherteks \code{KHOORZRUOG} dan cipherteks \code{SCRAM}. Serangan
          apa yang paling tepat untuk masing-masing, bila untuk yang kedua penyerang
          berhak memilih plainteksnya sendiri? Tunjukkan langkahnya.
    \item Sebuah sistem mengirim perintah transfer tanpa nomor urut dan tanpa cap
          waktu. Kunci tetap rahasia dan ciphernya tidak dipecahkan, tetapi penyerang
          tetap dapat merugikan bank. Serangan apa yang ia pakai, dan bagaimana
          penangkalannya?
    \item Jelaskan mengapa hasil pemulihan kunci melalui analisis daya (\textit{DPA})
          tidak dapat dicegah hanya dengan memilih algoritma yang lebih kuat.
\end{enumerate}
\end{obeassessment}
