\section{Contoh Terhitung}
\label{sec:contoh-ecc}
% Sumber: ECC-1 hlm. 19--23; ECC-2 hlm. 12, 33--34, 43, dan 49.

Contoh-contoh pada bagian ini mengikuti alur bab, dari aritmetika medan sampai
protokol yang lengkap. Setiap angka di bawah ini telah diperiksa dengan program
pada bagian praktikum.

\begin{example}[Aritmetika polinomial di dalam medan $GF(2^8)$]
\label{ex:gf2m}
Setiap \textit{byte} dipandang sebagai polinom berderajat paling tinggi tujuh,
dengan setiap bit menjadi koefisien. Dengan kesepakatan itu,
\[
\code{0D} = x^3 + x^2 + 1,
\qquad
\code{06} = x^2 + x .
\]

\textbf{(a) Penjumlahan.} Penjumlahan di dalam $GF(2)$ dilakukan modulo dua,
sehingga $1 + 1 = 0$. Karena itu penjumlahan dua polinom sama dengan operasi
XOR bit demi bit. Untuk polinom di atas,
\[
(x^3 + x^2 + 1) \oplus (x^2 + x) = x^3 + x + 1 = \code{0B} .
\]
Suku $x^2$ lenyap karena muncul pada kedua polinom. Dengan cara yang sama,
\[
\code{57} \oplus \code{83} = \code{D4} ,
\]
sebab dalam biner $\code{57} = 01010111$, $\code{83} = 10000011$, dan XOR
keduanya adalah $11010100 = \code{D4}$.

\textbf{(b) Perkalian.} Perkalian dilakukan dalam dua tahap: kalikan dahulu
seperti polinom biasa, lalu reduksi hasilnya dengan polinom reduksi
$x^8 + x^4 + x^3 + x + 1$ (yaitu $\code{11B}$, polinom yang dipakai AES).
Kedua faktor ditulis dalam bentuk polinom,
\[
\code{57} = x^6 + x^4 + x^2 + x + 1,
\qquad
\code{83} = x^7 + x + 1 .
\]
Karena $\code{83}$ memiliki bit pada posisi $7$, $1$, dan $0$, perkaliannya
adalah XOR dari tiga salinan $\code{57}$ yang digeser sejauh posisi bit
tersebut. Tabel~\ref{tab:gf2m-kali} memperlihatkan langkah itu.

\begin{table}[htbp]
    \centering
    \caption{Perkalian dua polinom di dalam $GF(2^8)$: hasil kali mentah
    berderajat $13$, lalu direduksi menjadi berderajat $7$}
    \label{tab:gf2m-kali}
    \begin{tabular}{@{}l l l@{}}
    \toprule
    \textbf{Langkah} & \textbf{Biner} & \textbf{Bentuk polinom} \\
    \midrule
    $\code{57} \ll 0$        & $01010111$          & $x^6 + x^4 + x^2 + x + 1$ \\
    \addlinespace
    $\code{57} \ll 1$        & $10101110$          & $x^7 + x^5 + x^3 + x^2 + x$ \\
    \addlinespace
    $\code{57} \ll 7$        & $10101110000000$    & $x^{13} + x^{11} + x^9 + x^8 + x^7$ \\
    \midrule
    XOR ketiganya            & $10101101111001$    & $x^{13} + x^{11} + x^9 + x^8 + x^6
                             + x^5 + x^4 + x^3 + 1$ \\
    \midrule
    $\oplus\,(\code{11B} \ll 5)$ & $0000100000011001$ & $x^{11} + x^4 + x^3 + 1$ \\
    \addlinespace

    \bottomrule
    \end{tabular}

Baris keempat Tabel~\ref{tab:gf2m-kali} adalah hasil kali mentah. Hasil itu
berderajat $13$, sehingga masih di luar medan. Dua baris terakhir
memperlihatkan reduksinya. Hasil akhirnya adalah $x^7 + x^6 + 1$, yaitu
$\code{C1}$. Jadi
\[
\code{57} \cdot \code{83} = \code{C1} .
\]

\textbf{(c) Pemeriksaan pada medan yang lebih kecil.} Di dalam $GF(2^4)$ dengan
polinom reduksi $x^4 + x + 1$,
\[
(x^3 + x^2 + 1)(x^2 + x) = x^5 + x^3 + x^2 + x .
\]
Hasil kali itu berderajat lima, sehingga direduksi satu langkah:
\[
(x^5 + x^3 + x^2 + x) \oplus \bigl(x \cdot (x^4 + x + 1)\bigr)
= (x^5 + x^3 + x^2 + x) \oplus (x^5 + x^2 + x) = x^3 .
\]
Sisanya berderajat tiga, jadi hasil akhirnya adalah $x^3 = \code{1000}$.

% Koreksi: ECC-1 hlm. 23 menulis hasil kali mentah 10110; bentuk binernya
% 101110. ECC-1 hlm. 18 menyebut x^2 + 1 tak tereduksi; padahal
% x^2 + 1 = (x + 1)^2. Faktorisasi yang benar untuk x^5 + x^2 + x + 1
% adalah (x^2 + 1)(x^3 + x + 1).

Dua angka pada berkas sumber perlu dikoreksi. Pertama, hasil kali mentah di
atas ditulis sebagai $\code{10110}$, padahal bentuk binernya adalah
$\code{101110}$. Kedua, berkas sumber menyebut $x^2 + 1$ tidak dapat
direduksi, padahal di dalam $GF(2)$ polinom itu sama dengan $(x+1)^2$. Karena
itu faktorisasi $x^5 + x^2 + x + 1$ pada berkas sumber juga salah. Faktorisasi
yang benar adalah
\[
x^5 + x^2 + x + 1 = (x^2 + 1)(x^3 + x + 1) .
\]
\end{example}

\begin{example}[Penjumlahan titik pada kurva di atas $GF(11)$]
\label{ex:gf11}
Ambil kurva
\[
y^2 \equiv x^3 + x + 6 \pmod{11},
\]
yang ordonya $13$ dan telah dienumerasi pada
Subbagian~\ref{subsec:enumerasi-gf-p}. Dua di antara titiknya adalah $P(2, 4)$
dan $Q(5, 9)$. Keduanya memenuhi kurva, sebab $4^2 = 16 \equiv 5$ dan
$2^3 + 2 + 6 = 16 \equiv 5$ untuk $P$, serta $9^2 = 81 \equiv 4$ dan
$5^3 + 5 + 6 = 136 \equiv 4$ untuk $Q$.

\textbf{(a) Penjumlahan $P + Q$.} Dengan $a = 1$ dan $p = 11$, gradiennya
menurut Persamaan~\eqref{eq:gradien-mod-p} adalah
\[
m \equiv (9 - 4)(5 - 2)^{-1} \equiv 5 \cdot 3^{-1} \pmod{11} .
\]
Karena $3 \cdot 4 = 12 \equiv 1$, invers dari $3$ adalah $4$, sehingga
$m \equiv 5 \cdot 4 = 20 \equiv 9$. Selanjutnya
\[
x_r \equiv m^2 - x_p - x_q \equiv 81 - 2 - 5 = 74 \equiv 8 \pmod{11},
\]
\[
y_r \equiv m(x_p - x_r) - y_p \equiv 9(2 - 8) - 4 = -58 \equiv 8 \pmod{11},
\]
sebab $-58 + 6 \cdot 11 = 8$. Jadi $P + Q = (8, 8)$.

\textbf{(b) Penggandaan $2P$.} Gradiennya diganti menurut
Persamaan~\eqref{eq:gradien-penggandaan}:
\[
m \equiv (3 \cdot 2^2 + 1)(2 \cdot 4)^{-1} \equiv 13 \cdot 8^{-1}
\pmod{11} .
\]
Karena $8 \cdot 7 = 56 \equiv 1$, invers dari $8$ adalah $7$, sehingga
$m \equiv 13 \cdot 7 = 91 \equiv 3$. Koordinat hasilnya
\[
x_r \equiv m^2 - 2x_p \equiv 9 - 4 = 5, \qquad
y_r \equiv m(x_p - x_r) - y_p \equiv 3(2 - 5) - 4 = -13 \equiv 9 ,
\]
jadi $2P = (5, 9)$.

Hasil itu sama dengan $Q$. Artinya $Q$ adalah kelipatan dari $P$, sehingga
$P + Q = 3P = (8, 8)$. Tabel pelelaran $kP$ untuk $k = 1, 2, \dots, 13$ pada
kurva ini diperlihatkan pada Tabel~\ref{tab:pelelaran-gf11}.
\end{example}

\begin{example}[Penjumlahan dan penggandaan titik di atas bilangan riil]
\label{ex:real}
Kurva di atas bilangan riil dipakai untuk memahami geometrinya, bukan untuk
kriptografi. Ambil kurva $y^2 = x^3 + 2x + 4$ dengan $a = 2$ dan $b = 4$.
Titik $P(2, 4)$ memenuhi kurva karena $4^2 = 8 + 4 + 4 = 16$. Titik $Q(0, 2)$
juga memenuhi karena $2^2 = 0 + 0 + 4 = 4$.

Untuk $P + Q$, gradien garis $PQ$ adalah
\[
m = \frac{4 - 2}{2 - 0} = 1 ,
\]
sehingga $x_r = 1 - 2 - 0 = -1$ dan $y_r = 1 \cdot (2 + 1) - 4 = -1$. Jadi
$P + Q = (-1, -1)$. Titik itu benar pada kurva sebab
$(-1)^2 = 1 = -1 - 2 + 4$.

Penggandaannya menghasilkan pecahan:
\[
m = \frac{3(2)^2 + 2}{2(4)} = \frac{14}{8} = \frac{7}{4},
\]
\[
x_r = \left(\frac{7}{4}\right)^2 - 2(2) = \frac{49}{16} - 4 = -\frac{15}{16},
\]
\[
y_r = \frac{7}{4}\left(2 + \frac{15}{16}\right) - 4
    = \frac{7 \cdot 47}{64} - 4 = \frac{73}{64} .
\]
Jadi $2P = \left(-\frac{15}{16}, \frac{73}{64}\right)$, atau
$(-0{,}9375;\ 1{,}140625)$. Bandingkan dengan kurva di atas $GF(11)$: di sana
koordinat hasilnya tetap bilangan bulat, sedangkan di sini koordinatnya
menjadi pecahan. Itulah sebabnya kriptografi memakai kurva di atas medan
berhingga.
\end{example}

\begin{example}[Pertukaran kunci ECDH]
\label{ex:ecdh}
Ambil kurva
\[
y^2 \equiv x^3 + 2x + 1 \pmod{5},
\]
yang memiliki $6$ titik berhingga ditambah $\mathcal{O}$, sehingga orderya
$7$. Titik basis yang disepakati adalah $B(0, 1)$, yang memenuhi kurva sebab
$1^2 = 0 + 0 + 1$. Alice memilih kunci privat $a = 2$ dan Bob $b = 3$. Kunci
publik masing-masing adalah
\[
P_A = 2B = (1, 3),
\qquad
P_B = 3B = (3, 3).
\]
Setiap pihak menghitung kunci bersama dari kunci publik pihak lain:
\[
K = a \cdot P_B = 2(3, 3) = (0, 4),
\qquad
K = b \cdot P_A = 3(1, 3) = (0, 4) .
\]
Keduanya memperoleh titik $(0, 4)$. Titik itu sama dengan $-B$, sebab $B$
berorde $7$ sehingga $6B = -B$.
\end{example}

\begin{example}[Pengkodean pesan dengan metode Koblitz]
\label{ex:koblitz}
Agar pesan dapat dienkripsi dengan ECEG, pesan harus diubah menjadi titik pada
kurva. Metode Koblitz melakukan itu dengan mencoba beberapa nilai $x$.

Ambil kurva
\[
y^2 \equiv x^3 - x + 188 \pmod{751},
\]
yang memiliki $N = 727$ titik. Huruf \texttt{A} sampai \texttt{Z} bernilai
$10$ sampai $35$, sehingga huruf \texttt{B} bernilai $m = 11$. Parameter
basisnya $k = 20$, sehingga nilai $x$ pertama yang dicoba adalah
\[
x = mk + 1 = (11)(20) + 1 = 221
\]
menurut Persamaan~\eqref{eq:koblitz-x}. Sulihkan ke dalam kurva:
\[
221^3 - 221 + 188 = 10\,793\,861 - 33 = 10\,793\,828
\equiv 456 \pmod{751} .
\]
Nilai $456$ bukan kuadrat modulo $751$, sehingga $x = 221$ tidak memberi titik.
Pencarian berlanjut seperti pada Tabel~\ref{tab:koblitz-b}: $x = 222$ memberi
$446$ dan $x = 223$ memberi $266$, keduanya juga bukan kuadrat. Baru pada
percobaan keempat berhasil. Untuk $x = 224$ diperoleh
\[
224^3 - 224 + 188 = 11\,239\,424 - 36 = 11\,239\,388
\equiv 673 \pmod{751} ,
\]
dan $673$ adalah kuadrat modulo $751$. Dua akarnya adalah $y = 248$ dan
$y = 503$, sebab $248^2 = 61\,504 = 81 \cdot 751 + 673$ dan
$503^2 = 253\,009 = 336 \cdot 751 + 673$. Jadi huruf \texttt{B} dikodekan
menjadi titik $(224, 248)$ pada kurva tersebut.

Penerima memulihkan pesannya dengan Persamaan~\eqref{eq:koblitz-dekode},
\[
m = \left\lfloor \frac{224 - 1}{20} \right\rfloor
  = \left\lfloor \frac{223}{20} \right\rfloor = 11 ,
\]
yang memang nilai untuk huruf \texttt{B}.
\end{example}
