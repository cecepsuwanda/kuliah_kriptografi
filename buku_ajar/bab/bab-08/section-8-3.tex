\section{TLS 1.3 dan Protokol Keamanan Modern}

\subsection{Evolusi TLS dari Versi 1.2 ke 1.3}

TLS 1.3 (RFC 8446, 2018) merupakan revisi besar dari protokol TLS yang menyederhanakan handshake dan meningkatkan keamanan \cite{rfc8446}. TLS 1.3 menghilangkan banyak fitur yang tidak aman dari versi sebelumnya dan fokus pada cipher suites modern.

\textbf{Perubahan Utama TLS 1.3}:
\begin{itemize}
  \item \textbf{Handshake disederhanakan}: Dari 2-RTT menjadi 1-RTT (opsional 0-RTT)
  \item \textbf{Cipher suites terbatas}: Hanya AEAD (AES-GCM, ChaCha20-Poly1305)
  \item \textbf{Perfect Forward Secrecy}: Wajib untuk semua koneksi
  \item \textbf{Enkripsi semua handshake}: Tidak ada plaintext setelah ClientHello
\end{itemize}

\begin{table}[htbp]
  \centering
  \small
  \begin{tabular}{lcc}
    \toprule
    Fitur & TLS 1.2 & TLS 1.3 \\
    \midrule
    Round Trip Time & 2-RTT & 1-RTT (opsional 0-RTT) \\
    Cipher Suites & 37+ & 5 \\
    PFS & Opsional & Wajib \\
    Static RSA & Diperbolehkan & Dihapus \\
    Renegotiation & Didukung & Dihapus \\
    \bottomrule
  \end{tabular}
  \caption{Perbandingan TLS 1.2 vs TLS 1.3.}
\end{table}

\subsection{TLS 1.3 Handshake Process}

TLS 1.3 handshake dirancang untuk efisiensi dan keamanan maksimal.

\textbf{Langkah-langkah Handshake}:
\begin{enumerate}
  \item \textbf{ClientHello}: Mengirim supported groups, cipher suites, key share
  \item \textbf{ServerHello}: Memilih cipher suite, key share, dan mengirim certificate
  \item \textbf{EncryptedExtensions}: Konfigurasi protokol (ALPN, max early data)
  \item \textbf{Certificate}: Sertifikat server (opsional untuk client auth)
  \textbf{CertificateVerify}: Bukti kepemilikan private key
  \item \textbf{Finished}: Pertukaran selesai, aplikasi data dapat dikirim
\end{enumerate}

\subsubsection{0-RTT Handshake (Resumption)}

0-RTT memungkinkan client mengirim data aplikasi pada pesan pertama jika telah memiliki koneksi sebelumnya.

\begin{lstlisting}[style=cstyle, caption={TLS 1.3 0-RTT Implementation}]
// Client side - 0-RTT data
if (session_resumed && early_data_accepted) {
    // Send 0-RTT application data
    send_early_data(application_data);
}

// Server side - 0-RTT handling
void handle_early_data(uint8_t *data, size_t len) {
    if (anti_replay_check(data)) {
        // Process early data
        process_application_data(data, len);
    } else {
        // Reject replayed data
        send_alert(UNEXPECTED_MESSAGE);
    }
}
\end{lstlisting}

\textbf{Alasan Pembatasan Cipher Suite}:
\begin{itemize}
  \item \textbf{Keamanan}: Hanya cipher dengan bukti keamanan kuat
  \item \textbf{Efisiensi}: AEAD menyediakan enkripsi dan autentikasi dalam satu operasi
  \item \textbf{Simplifikasi}: Mengurangi kompleksitas implementasi dan audit
\end{itemize}

\subsection{Key Schedule TLS 1.3}

TLS 1.3 menggunakan HKDF (HMAC-based Extract-and-Expand Key Derivation Function) untuk derivasi kunci:

\begin{lstlisting}[style=cstyle, caption={TLS 1.3 Key Schedule}]
typedef struct {
    uint8_t secret[32];
    uint8_t key[32];
    uint8_t iv[12];
} TrafficKeys;

// HKDF-Extract
void hkdf_extract(uint8_t *salt, uint8_t *ikm, uint8_t *prk) {
    hmac_sha256(salt, 32, ikm, 32, prk);
}

// HKDF-Expand
void hkdf_expand(uint8_t *prk, uint8_t *info, size_t info_len, 
                 uint8_t *okm, size_t L) {
    uint8_t t[32];
    size_t t_len = 0;
    uint8_t counter = 1;
    
    while (t_len < L) {
        uint8_t input[32 + info_len + 1];
        memcpy(input, t, t_len);
        memcpy(input + t_len, info, info_len);
        input[t_len + info_len] = counter++;
        
        hmac_sha256(prk, 32, input, t_len + info_len + 1, t);
        
        size_t copy_len = (L - t_len < 32) ? (L - t_len) : 32;
        memcpy(okm + t_len, t, copy_len);
        t_len += copy_len;
    }
}

// Derive traffic keys
void derive_traffic_keys(uint8_t *master_secret, TrafficKeys *keys) {
    uint8_t key_label[] = "tls13 c ap traffic";
    uint8_t iv_label[] = "tls13 c iv";
    
    hkdf_expand(master_secret, key_label, sizeof(key_label) - 1, 
                keys->key, 32);
    hkdf_expand(master_secret, iv_label, sizeof(iv_label) - 1, 
                keys->iv, 12);
}
\end{lstlisting}

\subsection{Perfect Forward Secrecy (PFS)}

TLS 1.3 mewajibkan PFS untuk semua koneksi menggunakan Diffie-Hellman (Ephemeral) atau Elliptic Curve Diffie-Hellman:

\begin{lstlisting}[style=cstyle, caption={ECDHE Key Exchange dalam TLS 1.3}]
typedef struct {
    uint8_t private_key[32];
    uint8_t public_key[65]; // Uncompressed format
} ECKeyPair;

// Generate ECDHE key pair
void generate_ecdhe_keypair(ECKeyPair *keypair) {
    // Generate private key
    random_bytes(keypair->private_key, 32);
    
    // Compute public key: Q = d * G
    scalar_multiply(G, keypair->private_key, keypair->public_key);
}

// Compute shared secret
void compute_shared_secret(uint8_t *private_key, uint8_t *peer_public_key, 
                          uint8_t *shared_secret) {
    // Compute: shared = d * Q_peer
    ECPoint peer_point = decompress_public_key(peer_public_key);
    ECPoint shared_point = scalar_multiply(peer_point, private_key);
    compress_point(shared_point, shared_secret);
}

// TLS 1.3 handshake with ECDHE
void tls13_handshake_ecdhe() {
    ECKeyPair client_key, server_key;
    uint8_t client_shared[32], server_shared[32];
    
    // Client generates key pair
    generate_ecdhe_keypair(&client_key);
    
    // Server generates key pair
    generate_ecdhe_keypair(&server_key);
    
    // Exchange public keys
    send_message("ClientKeyShare", client_key.public_key, 65);
    send_message("ServerKeyShare", server_key.public_key, 65);
    
    // Compute shared secrets
    compute_shared_secret(client_key.private_key, server_key.public_key, 
                        client_shared);
    compute_shared_secret(server_key.private_key, client_key.public_key, 
                        server_shared);
    
    // Both should get the same shared secret
    assert(memcmp(client_shared, server_shared, 32) == 0);
}
\end{lstlisting}

\subsection{Implementasi TLS 1.3 dalam C}

\begin{lstlisting}[style=cstyle, caption={TLS 1.3 Client Implementation}]
#include <stdint.h>
#include <string.h>

typedef struct {
    uint8_t version[2];
    uint8_t random[32];
    uint8_t cipher_suites[2];
    uint8_t extensions[100];
} ClientHello;

typedef struct {
    uint8_t version[2];
    uint8_t random[32];
    uint8_t cipher_suite[2];
    uint8_t extensions[50];
} ServerHello;

// TLS 1.3 client handshake
int tls13_client_handshake(int socket_fd) {
    uint8_t buffer[4096];
    ClientHello ch;
    ServerHello sh;
    
    // 1. Send ClientHello
    build_client_hello(&ch);
    send_tls_message(socket_fd, CLIENT_HELLO, &ch, sizeof(ch));
    
    // 2. Receive ServerHello
    receive_tls_message(socket_fd, SERVER_HELLO, &sh, sizeof(sh));
    
    // 3. Receive remaining handshake messages
    receive_encrypted_extensions(socket_fd);
    receive_certificate(socket_fd);
    receive_certificate_verify(socket_fd);
    receive_finished(socket_fd);
    
    // 4. Send Finished
    send_finished(socket_fd);
    
    // 5. Handshake complete, can send application data
    return 0;
}

// Build ClientHello
void build_client_hello(ClientHello *ch) {
    ch->version[0] = 0x03; // TLS 1.3 version
    ch->version[1] = 0x04;
    
    // Generate random
    random_bytes(ch->random, 32);
    
    // Set supported cipher suites
    ch->cipher_suites[0] = 0x13; // TLS_AES_128_GCM_SHA256
    ch->cipher_suites[1] = 0x01;
    
    // Add extensions (key_share, supported_groups, etc.)
    build_extensions(ch->extensions);
}
\end{lstlisting}

\subsection{Keamanan TLS 1.3}

\textbf{Improvements Keamanan}:
\begin{itemize}
  \item \textbf{Encrypted Handshake}: Semua pesan setelah ClientHello dienkripsi
  \item \textbf{No Renegotiation}: Menghilangkan vektor serangan renegotiation
  \item \textbf{Stronger Cipher Suites}: Hanya AEAD dengan keamanan terbukti
  \item \textbf{Mandatory PFS}: Tidak ada kunci statik yang dapat dikompromikan
\end{itemize}

\textbf{Serangan yang Dihilangkan}:
\begin{itemize}
  \item \textbf{ROBOT Attack}: Memanfaatkan RSA PKCS\#1 v1.5
  \item \textbf{POODLE}: Memanfaatkan CBC mode
  \item \textbf{DROWN}: Memanfaatkan SSLv2 fallback
  \item \textbf{Logjam}: Memanfaatkan weak DH parameters
\end{itemize}

\subsection{0-RTT Security Considerations}

0-RTT menyediakan latency reduction tetapi memiliki trade-off keamanan:

\begin{itemize}
  \item \textbf{Replay Attacks}: Penyerang dapat memutar ulang 0-RTT data
  \item \textbf{Forward Secrecy}: 0-RTT data tidak memiliki PFS
  \item \textbf{Mitigasi}: Server harus melakukan anti-replay checks
\end{itemize}

\begin{lstlisting}[style=cstyle, caption={Anti-Replay Protection}]
#define REPLAY_WINDOW_SIZE 64

typedef struct {
    uint64_t bitmap; // Bitmask untuk replay detection
    uint64_t next_expected;
} ReplayDetector;

int check_replay(ReplayDetector *detector, uint64_t seq_num) {
    if (seq_num < detector->next_expected - REPLAY_WINDOW_SIZE) {
        return 1; // Too old, reject
    }
    
    if (seq_num >= detector->next_expected) {
        // New sequence number
        detector->bitmap <<= (seq_num - detector->next_expected);
        detector->next_expected = seq_num + 1;
        return 0; // Accept
    }
    
    // Check if already seen
    uint64_t bit_pos = detector->next_expected - seq_num - 1;
    if (detector->bitmap & (1ULL << bit_pos)) {
        return 1; // Already seen, reject
    }
    
    // Mark as seen
    detector->bitmap |= (1ULL << bit_pos);
    return 0; // Accept
}
\end{lstlisting}

\subsection{Praktikum: TLS 1.3 Implementation}

\begin{lstlisting}[style=cstyle, caption={Simple TLS 1.3 Echo Server}]
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>

int main() {
    int server_fd, client_fd;
    struct sockaddr_in address;
    int opt = 1;
    int addrlen = sizeof(address);
    
    // Create socket
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(8443);
    
    bind(server_fd, (struct sockaddr*)&address, sizeof(address));
    listen(server_fd, 3);
    
    printf("TLS 1.3 Echo Server listening on port 8443\n");
    
    while (1) {
        client_fd = accept(server_fd, (struct sockaddr*)&address, (socklen_t*)&addrlen);
        
        if (fork() == 0) {
            close(server_fd);
            
            // Perform TLS 1.3 handshake
            if (tls13_server_handshake(client_fd) == 0) {
                printf("Handshake successful\n");
                
                // Echo loop
                uint8_t buffer[1024];
                int bytes_read;
                while ((bytes_read = receive_application_data(client_fd, buffer, 1024)) > 0) {
                    send_application_data(client_fd, buffer, bytes_read);
                }
            }
            
            close(client_fd);
            exit(0);
        }
        
        close(client_fd);
    }
    
    return 0;
}
\end{lstlisting}

\subsection{Library dan Implementasi}

\textbf{Library yang Mendukung TLS 1.3}:
\begin{itemize}
  \item \textbf{OpenSSL 1.1.1+}: Implementasi lengkap TLS 1.3
  \item \textbf{BoringSSL}: Fork Google dengan optimasi
  \item \textbf{wolfSSL}: Library C ringan untuk embedded
  \item \textbf{mbedTLS}: Library ARM untuk IoT
\end{itemize}

\begin{catatan}
TLS 1.3 merupakan standar keamanan internet modern. Implementasi TLS 1.3 dari contoh di atas adalah untuk tujuan edukasi. Untuk produksi, gunakan library yang telah diaudit keamanannya seperti OpenSSL atau BoringSSL.
\end{catatan}
