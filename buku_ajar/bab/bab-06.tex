\begin{rangkuman}
Bab ini membahas kriptografi asimetris dan algoritma RSA: pembangkitan kunci, enkripsi, dekripsi, padding modern, hybrid encryption, serta implementasi C.

\textbf{Poin Kunci:}
\begin{itemize}
  \item RSA: $n=pq$, $\varphi(n)=(p-1)(q-1)$, $ed \equiv 1 \pmod{\varphi(n)}$
  \item Enkripsi: $c = m^e \bmod n$; Dekripsi: $m = c^d \bmod n$
  \item OAEP direkomendasikan untuk enkripsi, PSS untuk tanda tangan
  \item Skema hybrid (KEM-DEM) memakai RSA untuk kunci dan AES untuk data
  \item Keamanan bergantung pada kesulitan memfaktorkan $n$ dan kualitas implementasi
\end{itemize}

\textbf{Kata Kunci}: \crypt{RSA}, \crypt{OAEP}, \crypt{PSS}, \crypt{Hybrid Encryption}, \crypt{KEM-DEM}, \crypt{CRT}, \crypt{Kunci Publik}
\end{rangkuman}
