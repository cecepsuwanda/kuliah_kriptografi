\end{subcpmk}

\noindent\textit{Rujukan:} Munir slide 03--06 (Kriptografi Klasik bagian 1 dan 2, One-Time Pad, Serangan terhadap Kriptografi) \cite{munir_slides_itb}; Stallings \cite{stallings2017}; Schneier \cite{schneier1996}; Stinson \cite{stinson2018}; Shannon \cite{shannon1949}.

Bab ini mengikuti alur slide kuliah: dimulai dari Caesar cipher dan cipher transposisi, berlanjut ke cipher abjad-tunggal beserta analisis frekuensi, cipher abjad-majemuk (Vigen\`{e}re), cipher poligram (Playfair dan Hill), lalu One-Time Pad sebagai satu-satunya cipher yang aman sempurna, dan mesin Enigma. Bab ditutup dengan pengelompokan jenis-jenis serangan terhadap kriptografi dan ringkasan kriptanalisis semua cipher klasik. Subbagian yang ditandai \textbf{pengayaan} (kriptanalisis langkah demi langkah, varian Vigen\`{e}re, dan Enigma) tidak wajib untuk mencapai Sub-CPMK 1.1, tetapi sangat dianjurkan.

\begin{latihan}
  \item Enkripsi kalimat \texttt{ADA RENCANA PENYELUNDUPAN NARKOBA DI BANDARA} dengan Caesar cipher $k=5$. Dekripsi kembali hasilnya.
  \item Enkripsi \texttt{sistem dan teknologi informasi itb} dengan columnar transposition berkata kunci \texttt{TOMBAK}, lalu enkripsi ulang hasilnya dengan Caesar $k=3$ (super-enkripsi).
  \item Enkripsi \texttt{HELLO} dengan Vigen\`{e}re cipher berkunci \texttt{KEY}. Tunjukkan perhitungan setiap huruf.
  \item Dengan matriks kunci Playfair \texttt{JALAN GANESHA SEPULUH}, enkripsi plainteks \texttt{jangan lupa bawa buku}. Sebutkan aturan yang dipakai untuk setiap bigram.
  \item Enkripsi \texttt{help} dengan Hill cipher $2 \times 2$ berkunci $\mathbf{K} = \begin{pmatrix} 3 & 10 \\ 15 & 9 \end{pmatrix}$, lalu dekripsi kembali memakai $\mathbf{K}^{-1}$.
  \item Ubahlah program Vigen\`{e}re cipher dalam Python (Kode~\ref{lst:vigenere-py}) menjadi Extended Vigen\`{e}re (modulo 256) sehingga dapat mengenkripsi berkas teks.
  \item Mengapa One-Time Pad disebut aman sempurna, tetapi tidak praktis?
  \item Diberikan cipherteks OTP \texttt{TLCYKUMGDFAWTZVOYKLENSZZHYZRWQ}. Temukan kunci yang menghasilkan plainteks \texttt{mr johnson left his house last night}, lalu temukan kunci lain yang menghasilkan plainteks \texttt{i saw the mysterious plane behind me}. Apa makna hasil ini bagi kriptanalis?
  \item Seorang operator memakai kunci OTP yang sama untuk dua pesan. Tunjukkan bahwa $(C_1 - C_2) \bmod 26 = (P_1 - P_2) \bmod 26$, lalu jelaskan mengapa \textit{two-time pad} tidak aman.
  \item Tentukan model serangan pada setiap skenario berikut:
  \begin{enumerate}
    \item penyerang menyadap surel terenkripsi yang selalu diawali ``Dengan hormat'';
    \item penyerang mengetikkan PIN pilihannya ke ATM sambil menyadap cipherteks yang dikirim ke bank;
    \item penyerang hanya merekam cipherteks dari kanal radio.
  \end{enumerate}
  \item Jelaskan tiga kriteria algoritma yang aman secara komputasi. Mengapa AES dianggap aman walaupun tidak memiliki \textit{perfect secrecy}?
  \item \textbf{Refleksi}: Substitusi monoalfabetik memiliki $26! \approx 2^{88}$ kunci, jauh lebih banyak daripada kunci DES ($2^{56}$). Mengapa ruang kunci yang besar tidak menjamin keamanan?
\end{latihan}

\begin{asesmen}
\textbf{Instrumen Penilaian untuk Sub-CPMK 1.1}

\textbf{A. Pilihan Ganda}

\begin{enumerate}
  \item Caesar cipher adalah kasus khusus dari:
  \begin{enumerate}
    \item Block cipher
    \item Stream cipher
    \item Substitusi monoalfabetik
    \item Transposisi
  \end{enumerate}

  \item Serangan yang efektif untuk memecahkan Caesar cipher:
  \begin{enumerate}
    \item Kasiski examination
    \item Brute force atau frequency analysis
    \item Differential cryptanalysis
    \item Birthday attack
  \end{enumerate}

  \item Pada Playfair cipher, huruf yang dibuang dari matriks kunci $5 \times 5$ adalah:
  \begin{enumerate}
    \item Q
    \item X
    \item J
    \item Z
  \end{enumerate}

  \item Histogram frekuensi huruf cipherteks yang cenderung datar merupakan ciri:
  \begin{enumerate}
    \item Caesar cipher
    \item Cipher abjad-majemuk seperti Vigen\`{e}re
    \item Cipher transposisi
    \item Substitusi monoalfabetik
  \end{enumerate}

  \item One-Time Pad memiliki \textit{perfect secrecy} hanya jika kuncinya:
  \begin{enumerate}
    \item Dibangkitkan dengan \textit{pseudo-random generator} dari \textit{seed} rahasia
    \item Acak sejati, sepanjang plainteks, dan dipakai sekali
    \item Diambil dari teks novel yang panjang
    \item Sepanjang 128 bit dan diulang secara periodik
  \end{enumerate}

  \item Penyerang memasukkan plainteks \texttt{AAAAAAAA} ke sistem enkripsi Vigen\`{e}re dan memperoleh kuncinya langsung dari cipherteks. Serangan ini termasuk:
  \begin{enumerate}
    \item \textit{Ciphertext-only attack}
    \item \textit{Known-plaintext attack}
    \item \textit{Chosen-plaintext attack}
    \item \textit{Replay attack}
  \end{enumerate}
\end{enumerate}

\textbf{B. Uraian}

\begin{enumerate}
  \item Bedakan teknik substitusi dan teknik transposisi, dan berikan masing-masing satu contoh cipher.
  \item Jelaskan langkah enkripsi Vigen\`{e}re cipher dan mengapa cipher ini lebih tahan terhadap analisis frekuensi dibandingkan Caesar cipher.
  \item Jelaskan gagasan metode Kasiski untuk memperkirakan panjang kunci Vigen\`{e}re.
  \item Mengapa Affine cipher dan Hill cipher mudah dipecahkan dengan \textit{known-plaintext attack}?
  \item Jelaskan dua alasan mengapa OTP tidak dapat dipecahkan, serta dua alasan mengapa OTP jarang dipakai dalam praktik.
  \item Bedakan \textit{key-recovery}, \textit{plaintext-recovery}, \textit{forgery}, dan \textit{replay attack}, masing-masing dengan satu contoh.
\end{enumerate}
\end{asesmen}

\begin{rangkuman}
Bab ini membahas algoritma kriptografi klasik yang disusun dari dua teknik dasar, substitusi dan transposisi: Caesar cipher dan ROT13, cipher transposisi (columnar, rail fence) dan super-enkripsi, substitusi monoalfabetik, Affine, Vigen\`{e}re beserta variannya, Playfair, Hill, One-Time Pad, dan mesin Enigma. Untuk setiap cipher dibahas pula cara memecahkannya, lalu cara-cara pemecahan itu dikelompokkan menjadi model serangan yang juga berlaku untuk cipher modern.

\textbf{Poin Kunci:}
\begin{itemize}
  \item Caesar: $c = (p + k) \bmod 26$, hanya 26 kunci sehingga pecah oleh \textit{exhaustive key search}; ROT13 adalah Caesar dengan $k = 13$.
  \item Transposisi mengubah urutan huruf tanpa menggantinya; super-enkripsi menggabungkan substitusi dan transposisi.
  \item Cipher abjad-tunggal ($26!$ kunci) tetap pecah oleh analisis frekuensi karena statistik bahasa terbawa ke cipherteks.
  \item Affine: $C \equiv mP + b \pmod{26}$ dengan $\gcd(m, 26) = 1$; Hill: $\mathbf{C} = \mathbf{K}\mathbf{P} \bmod 26$ dengan $\mathbf{K}$ invertibel. Keduanya linear sehingga rentan \textit{known-plaintext attack}.
  \item Vigen\`{e}re (abjad-majemuk) meratakan histogram, tetapi panjang kuncinya dapat ditemukan dengan metode Kasiski atau \textit{index of coincidence}.
  \item Playfair mengenkripsi bigram dengan matriks $5 \times 5$ dan dapat dipecahkan dengan analisis frekuensi bigram.
  \item OTP (Vernam, 1917) aman sempurna jika kunci acak sejati, sepanjang pesan, dan sekali pakai; Shannon (1949) membuktikan $\Pr(P = p \mid C = c) = \Pr(P = p)$. OTP tidak praktis karena pembangkitan dan distribusi kuncinya; kunci yang dipakai ulang (\textit{two-time pad}) atau diambil dari teks bermakna merusak keamanannya.
  \item Enigma adalah cipher abjad-majemuk elektromekanik yang dipecahkan dengan \textit{Bombe}.
  \item Model serangan menurut informasi penyerang: \textit{ciphertext-only}, \textit{known-plaintext}, \textit{chosen-plaintext}, \textit{adaptive chosen-plaintext}, dan \textit{chosen-ciphertext}. Menurut tujuannya: \textit{key-recovery}, \textit{plaintext-recovery}, \textit{distinguishing}, \textit{forgery}, dan \textit{replay}. \textit{Side-channel attack} menyerang implementasi, bukan matematika.
  \item Cipher praktis dirancang agar aman secara komputasi: biaya dan waktu pemecahannya melampaui nilai dan masa kerahasiaan informasi.
\end{itemize}

\textbf{Kata Kunci}: \crypt{Caesar}, \crypt{ROT13}, \crypt{Transposisi}, \crypt{Super-enkripsi}, \crypt{Monoalfabetik}, \crypt{Affine}, \crypt{Vigen\`{e}re}, \crypt{Playfair}, \crypt{Hill}, \crypt{One-Time Pad}, \crypt{Perfect Secrecy}, \crypt{Enigma}, \crypt{Analisis Frekuensi}, \crypt{Kasiski}, \crypt{Ciphertext-Only Attack}, \crypt{Known-Plaintext Attack}, \crypt{Chosen-Plaintext Attack}, \crypt{Chosen-Ciphertext Attack}, \crypt{Computationally Secure}, \crypt{Side-Channel Attack}
\end{rangkuman}

\begin{transition}
Setelah mempelajari cipher klasik dan kelemahannya, kita siap untuk memasuki \textbf{Bab V: Block Cipher dan Kriptografi Simetris Modern}.

Dalam Bab V, kita akan melihat evolusi dari cipher klasik ke algoritma modern:
\begin{itemize}
  \item \textbf{Stream Cipher}: LFSR, RC4, A5/1 â€” evolusi dari cipher klasik berurutan
  \item \textbf{Block Cipher}: DES (Feistel) dan AES (SPN) â€” pendekatan terstruktur
  \item \textbf{Mode Operasi}: ECB, CBC, CFB, OFB, CTR â€” mengatasi keterbatasan block cipher
\end{itemize}

\textbf{Koneksi dengan Bab IV}:
\begin{itemize}
  \item \textbf{Masalah Caesar}: Ruang kunci kecil $\rightarrow$ \textbf{Solusi AES}: Ruang kunci $2^{128}$, $2^{192}$, $2^{256}$
  \item \textbf{Masalah monoalfabetik dan Vigen\`{e}re}: Statistik bahasa bocor ke cipherteks $\rightarrow$ \textbf{Solusi DES/AES}: \textit{diffusion} dan \textit{confusion}
  \item \textbf{Masalah Affine dan Hill}: Cipher linear sehingga rentan \textit{known-plaintext attack} $\rightarrow$ \textbf{Solusi AES}: S-box nonlinear dan operasi dalam $\mathrm{GF}(2^8)$
  \item \textbf{Gagasan super-enkripsi}: substitusi + transposisi $\rightarrow$ \textbf{Cipher modern}: putaran berulang substitusi dan permutasi (jaringan Feistel, SPN)
  \item \textbf{Masalah OTP}: kunci acak sepanjang pesan sulit dibangkitkan dan didistribusikan $\rightarrow$ \textbf{Solusi stream cipher}: kunci pendek direntangkan menjadi \textit{keystream} yang aman secara komputasi
  \item \textbf{Model serangan}: cipher klasik pecah oleh \textit{ciphertext-only} dan \textit{known-plaintext attack} $\rightarrow$ \textbf{Target cipher modern}: tetap aman bahkan terhadap \textit{chosen-plaintext} dan \textit{chosen-ciphertext attack}
\end{itemize}

\textbf{Matematika yang Akan Diterapkan}:
\begin{itemize}
  \item \textbf{Aljabar Abstrak (Bab III)}: Grup dan field dalam desain AES
  \item \textbf{Aritmetika Modular}: Operasi dalam $\mathrm{GF}(2^8)$ untuk AES
  \item \textbf{Teori Informasi}: Konsep diffusion dan confusion
\end{itemize}

\textbf{Preview}: Di Bab V, kita akan mempelajari bagaimana DES menggunakan struktur Feistel yang elegan, dan bagaimana AES menggunakan operasi dalam Galois Field untuk mencapai keamanan yang jauh superior dibandingkan cipher klasik. Kita juga akan memahami mengapa mode operasi seperti CBC diperlukan untuk mengenkripsi data yang lebih besar dari block size.

Setelah Bab V, Anda akan mengerti mengapa cipher klasik tidak lagi digunakan dalam aplikasi serius dan bagaimana prinsip desain modern mengatasi keterbatasan fundamental yang kita pelajari dalam Bab IV.
\end{transition}
