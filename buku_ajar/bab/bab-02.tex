\begin{aktivitas}
  \item \textbf{Diskusi Kelompok (urgensi)}: Identifikasi 5 aplikasi sehari-hari yang menggunakan kriptografi (misalnya: WhatsApp, perbankan online, e-commerce). Jelaskan ancaman apa yang dilindungi masing-masing, dan apakah proteksinya lebih ke \textit{at rest}, \textit{in transit}, atau keduanya.
  
  \item \textbf{Empat Case Alice--Bob}: Petakan Case~1--4 (penyadapan, penyamaran, manipulasi, penyangkalan) ke layanan keamanan dan satu mekanisme kriptografi yang relevan.
  
  \item \textbf{Analisis Kasus}: Diberikan skenario: Eve mencuri ciphertext dari komunikasi Alice dan Bob. Apa yang bisa dan tidak bisa dilakukan Eve? Jelaskan peran confidentiality, integrity, dan authenticity.
  
  \item \textbf{Demo Wireshark (di jaringan sendiri)}: Rekam lalu lintas saat masuk ke sebuah situs uji HTTP (misalnya \url{http://testphp.vulnweb.com}) dan saat membuka situs HTTPS. Bandingkan apa yang terlihat oleh penyadap pada keduanya, lalu kaitkan dengan serangan pasif \textit{interception} dan \textit{traffic analysis}.
  
  \item \textbf{Mind Mapping}: Buat mind map yang menghubungkan confidentiality, integrity, availability, non-repudiation dengan mekanisme kriptografi yang mendukungnya.
  
  \item \textbf{Perbandingan}: Buat tabel perbandingan antara kriptografi simetris dan asimetris untuk: enkripsi file, pertukaran kunci, tanda tangan digital. Tambahkan satu baris: kapan fungsi hash dipakai.
\end{aktivitas}

\begin{latihan}
  \item Jelaskan definisi plaintext, ciphertext, cipher, dan key! Berikan contoh konkret.
  
  \item Sebutkan dan jelaskan empat layanan keamanan informasi (confidentiality, integrity, availability, non-repudiation)! Berikan contoh ancaman yang melanggar masing-masing.
  
  \item Apa perbedaan utama antara kriptografi simetris dan asimetris? Kapan masing-masing lebih cocok digunakan?
  
  \item Jelaskan model komunikasi kriptografi (Alice, Bob, Eve). Apa peran masing-masing, dan berikan contoh Alice/Bob yang berupa mesin (bukan manusia)?
  
  \item Bedakan serangan pasif dan serangan aktif. Klasifikasikan interception, traffic analysis, interruption, modification, fabrication, dan replay, serta sebutkan layanan keamanan yang diancam masing-masing.
  
  \item Mengapa enkripsi saja tidak cukup untuk mencegah \textit{replay attack}? Mekanisme apa yang dapat mencegahnya?
  
  \item Jelaskan cara kerja \textit{man-in-the-middle attack} pada pertukaran kunci Diffie--Hellman. Apa yang diperlukan agar serangan ini gagal?
  
  \item Jelaskan \textbf{prinsip Kerckhoffs}. Mengapa algoritma modern dipublikasikan?
  
  \item Bedakan enkripsi, encoding, hashing, dan tokenisasi. Mana yang melindungi kerahasiaan plaintext?
  
  \item Apa perbedaan \textit{encryption at rest} dan \textit{encryption in transit}? Berikan satu contoh masing-masing.
  
  \item Sebutkan peran BSSN (atau lembaga sandi/siber terkait) dalam konteks kriptografi di Indonesia.
  
  \item Berikan 3 contoh ancaman keamanan dan jelaskan bagaimana kriptografi dapat mengatasinya.
  
  \item \textbf{Refleksi}: Bagaimana pemahaman Anda tentang keamanan informasi sebelum dan sesudah mempelajari bab ini?
\end{latihan}

\begin{asesmen}
\textbf{Instrumen Penilaian untuk Sub-CPMK 2.1, 2.2}

\textbf{A. Pilihan Ganda}

\begin{enumerate}
  \item Manakah yang BUKAN merupakan layanan keamanan informasi?
  \begin{enumerate}
    \item Confidentiality
    \item Integrity
    \item Complexity
    \item Non-repudiation
  \end{enumerate}
  
  \item Dalam kriptografi simetris, jumlah kunci yang digunakan adalah:
  \begin{enumerate}
    \item Satu kunci untuk enkripsi dan dekripsi
    \item Dua kunci berbeda
    \item Tidak ada kunci
    \item Kunci publik saja
  \end{enumerate}
  
  \item Fungsi utama enkripsi adalah untuk mencapai:
  \begin{enumerate}
    \item Integritas
    \item Kerahasiaan (confidentiality)
    \item Keaslian
    \item Ketersediaan
  \end{enumerate}
  
  \item Penyerang merekam pesan perintah transfer yang sah lalu mengirimkannya kembali keesokan harinya tanpa mendekripsinya. Serangan ini termasuk:
  \begin{enumerate}
    \item Serangan pasif berupa traffic analysis
    \item Serangan aktif berupa replay
    \item Serangan pasif berupa interception
    \item Serangan aktif berupa interruption
  \end{enumerate}
\end{enumerate}

\textbf{B. Essay}

\begin{enumerate}
  \item Jelaskan dengan kata-kata Anda sendiri apa yang dimaksud dengan kriptografi dan berikan 2 contoh penggunaannya dalam kehidupan nyata.
  
  \item Bandingkan kriptografi simetris dan asimetris. Mana yang lebih cocok untuk enkripsi file besar dan mengapa?
\end{enumerate}

\end{asesmen}

\begin{rangkuman}
Bab ini mengikuti alur: pengantar (definisi, sejarah, terminologi, Kerckhoffs, cryptosystem) $\to$ model komunikasi dan empat case $\to$ serangan pasif dan aktif $\to$ urgensi di TI $\to$ layanan keamanan $\to$ keluarga algoritma (simetris, asimetris, hash) $\to$ peta konsep dan tantangan masa depan.

\textbf{Poin Kunci:}
\begin{itemize}
  \item Kriptografi mengamankan informasi; kriptanalisis adalah ``lawannya''; keduanya membentuk kriptologi
  \item Kerckhoffs: algoritma boleh publik; yang rahasia adalah kunci
  \item Empat case Alice--Bob memotivasi confidentiality, authentication, integrity, non-repudiation
  \item Serangan pasif (interception, traffic analysis; lewat wiretapping, emisi elektromagnetik, atau akustik) sulit dideteksi sehingga dicegah dengan enkripsi; serangan aktif (interruption, modification, fabrication, replay, MITM) dideteksi dengan hash/MAC, tanda tangan digital, dan nonce/nomor urut
  \item Enkripsi $\neq$ encoding $\neq$ hashing $\neq$ tokenisasi
  \item Simetris: satu kunci, cepat; asimetris: pasangan kunci; hash: ringkasan satu arah untuk integritas
\end{itemize}

\textbf{Kata Kunci}: \crypt{Kriptografi}, \crypt{Kriptologi}, \crypt{Kerckhoffs}, \crypt{Plaintext}, \crypt{Ciphertext}, \crypt{Confidentiality}, \crypt{Integrity}, \crypt{Availability}, \crypt{Non-repudiation}, \crypt{Serangan Pasif}, \crypt{Serangan Aktif}, \crypt{Replay}, \crypt{Man-in-the-Middle}
\end{rangkuman}

\begin{transition}
Setelah memahami konsep dasar dan urgensi kriptografi di TI---termasuk prinsip Kerckhoffs, layanan keamanan, serta perbedaan simetris/asimetris/hash---kita siap untuk \textbf{Bab III: Matematika Kriptografi} (pengenalan). Setiap cipher yang akan dipelajari, dari Caesar hingga RSA dan AES, bertumpu pada matematika; Bab III membekali alat-alat tersebut.

Dalam Bab III (Sub-CPMK 2.3), kita mengenal:
\begin{itemize}
  \item Aritmetika modular, relatif prima, dan invers modulo
  \item Algoritma Euclid serta bilangan prima (Fermat, Euler)
  \item Teori informasi (entropi) --- mengapa cipherteks yang baik tampak acak
\end{itemize}
\end{transition}
