\begin{aktivitas}
  \item \textbf{Analisis Kasus}: Carilah berita mengenai kebocoran data pribadi yang terjadi di Indonesia (misal: BPJS, Tokopedia, atau BSI). Analisis layanan keamanan mana (CIA) yang dilanggar dan bagaimana peran kriptografi seharusnya bisa mencegah hal tersebut.
  \item \textbf{Simulasi Komunikasi}: Lakukan simulasi pengiriman pesan antara Alice dan Bob melalui perantara "Eve". Coba identifikasi perbedaan antara serangan pasif (hanya menguping) dan serangan aktif (mengubah isi pesan).
  \item \textbf{Diskusi Terminologi}: Diskusikan dalam kelompok mengenai perbedaan antara \textit{encoding}, \textit{hashing}, dan \textit{encryption}. Mengapa menganggap Base64 sebagai bentuk enkripsi adalah sebuah kesalahan fatal?
  \item \textbf{Eksplorasi Museum Sandi}: Jika memungkinkan, kunjungi Museum Sandi Yogyakarta atau akses situs resminya. Identifikasi satu alat sandi mekanik dan jelaskan secara sederhana bagaimana alat tersebut bekerja.
\end{aktivitas}

% LATIHAN DAN REFLEKSI

\begin{latihan}
  \item Jelaskan perbedaan antara \textit{Confidentiality}, \textit{Integrity}, dan \textit{Availability}. Berikan contoh skenario nyata di mana salah satu layanan tersebut dilanggar.
  \item Mengapa otentikasi (\textit{authentication}) sangat penting dalam komunikasi digital? Apa yang terjadi jika sebuah sistem hanya memiliki kerahasiaan tanpa otentikasi?
  \item Bedakan antara serangan pasif dan serangan aktif. Mana yang lebih mudah dideteksi oleh penerima pesan? Jelaskan alasannya.
  \item Jelaskan peran kunci (\textit{key}) dalam sebuah cipher. Mengapa algoritma cipher boleh diketahui publik namun kuncinya harus tetap rahasia?
  \item Apa perbedaan mendasar antara kriptografi klasik dan kriptografi modern dalam hal unit data yang diolah dan landasan keamanannya?
  \item Mengapa kriptanalisis dianggap sebagai "lawan" sekaligus "guru" bagi kriptografi?
\end{latihan}

\begin{asesmen}
\textbf{Instrumen Penilaian untuk Sub-CPMK 2.1, 2.2}

\textbf{A. Pilihan Ganda}

\begin{enumerate}
  \item Jika $\text{PBB}(a, n) = 1$, maka $a$ memiliki invers modulo $n$. Pernyataan ini adalah:
  \begin{enumerate}
    \item Benar
    \item Salah
    \item Benar hanya jika $n$ adalah bilangan prima
    \item Benar hanya jika $a < n$
  \end{enumerate}

  \item Nilai dari $-7 \pmod{11}$ adalah:
  \begin{enumerate}
    \item 4
    \item 7
    \item 8
    \item 1
  \end{enumerate}

  \item Entropi maksimum dari pesan yang hanya terdiri dari 26 huruf alfabet adalah sekitar:
  \begin{enumerate}
    \item 2.0 bit/karakter
    \item 4.7 bit/karakter
    \item 8.0 bit/karakter
    \item 1.0 bit/karakter
  \end{enumerate}
\end{enumerate}

\textbf{B. Essay}

\begin{enumerate}
  \item Jelaskan langkah-langkah Algoritma Euclidean Diperluas (EEA) untuk mencari invers modulo.
  \item Jelaskan peran fungsi Totien Euler $\varphi(n)$ dalam kriptografi asimetris.
\end{enumerate}
\end{asesmen}

\begin{rangkuman}
Bab ini membahas landasan teori dan konsep dasar kriptografi, mulai dari layanan keamanan informasi (CIA Triad), model komunikasi antara pengirim dan penerima, hingga ancaman yang dibawa oleh penyusup (\textit{intruder}). Kita juga telah membedakan antara terminologi dasar kriptografi, proses enkripsi-dekripsi, serta evolusi dari kriptografi klasik menuju kriptografi modern. Pemahaman konsep ini sangat krusial sebelum memasuki detail teknis algoritma dan matematika di bab-bab selanjutnya.
\end{rangkuman}
