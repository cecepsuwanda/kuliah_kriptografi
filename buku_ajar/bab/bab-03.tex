\begin{asesmen}
\textbf{Instrumen Penilaian untuk Sub-CPMK 2.3}

\textbf{A. Pilihan Ganda}

\begin{enumerate}
  \item Invers modular dari $a$ modulo $n$ hanya ada jika:
  \begin{enumerate}
    \item $a > n$
    \item $\gcd(a, n) = 1$
    \item $a$ prima
    \item $n$ prima
  \end{enumerate}
\end{enumerate}

\begin{rangkuman}
Bab ini membahas fondasi matematika kriptografi. Dimulai dari peta lima kelompok materi (teori bilangan, probabilitas, kompleksitas, teori informasi, aljabar abstrak), bab ini membahas aritmetika modular, relatif prima, dan invers modulo; algoritma Euclid dan Euclid diperluas; bilangan prima, teorema Fermat dan Euler, serta CRT; fast modular exponentiation; dan teori informasi (entropi). Contoh numerik merangkum semuanya, sedangkan aljabar abstrak (grup, ring, field, $\mathrm{GF}(2^n)$) disajikan sebagai pengayaan. Teori bilangan menjadi dasar RSA, entropi menjadi dasar analisis frekuensi, dan konsep field menjadi dasar AES.

\textbf{Poin Kunci:}
\begin{itemize}
  \item Kongruensi modular: $a \equiv b \pmod{n}$ jika $n \mid (a-b)$
  \item $a$ dan $b$ relatif prima jika $\gcd(a, b) = 1$; invers modular hanya ada jika $\gcd(a, n) = 1$
  \item Euclid diperluas: menghitung $\gcd$ sekaligus koefisien Bézout
  \item Chinese Remainder Theorem: menyelesaikan sistem kongruensi
  \item $\varphi(n) = (p-1)(q-1)$ untuk $n = pq$
  \item Fast exponentiation: $O(\log b)$ untuk menghitung $a^b \bmod n$
  \item Entropi $H(X) = -\sum p(x_i)\log_2 p(x_i)$, dengan $0 \le H \le \log_2 n$; ciphertext yang baik berentropi tinggi
  \item (Pengayaan) Grup, ring, field; $\mathrm{GF}(2^n)$ untuk operasi bit (misal AES)
\end{itemize}

{\raggedright \textbf{Kata Kunci}: \crypt{Aritmetika Modular}, \crypt{Kongruensi}, \crypt{Relatif Prima}, \crypt{Algoritma Euclid}, \crypt{Invers Modular}, \crypt{Chinese Remainder Theorem}, \crypt{Fast Exponentiation}, \crypt{Fungsi Euler}, \crypt{Entropi}, \crypt{Galois Field}\par}
\end{rangkuman}
