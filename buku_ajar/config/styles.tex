\usepackage{titlesec}

% --- Tata letak halaman --------------------------------------------------
\geometry{
    a4paper,
    left=3cm,
    right=2.5cm,
    top=3cm,
    bottom=3cm
}

% --- Format judul bab ----------------------------------------------------
% Disetel agar seragam dengan gaya bawaan kelas `book` (nomor bab pada
% baris tersendiri, lalu judul bab), sehingga tampilan buku tidak berubah.
% \chaptertitlename menghasilkan "Bab" untuk isi utama dan "Lampiran"
% untuk bagian lampiran -- keduanya dari babel indonesian.
\titleformat{\chapter}[display]
    {\normalfont\huge\bfseries}
    {\chaptertitlename~\thechapter}
    {20pt}
    {\Huge\bfseries}
\titlespacing*{\chapter}{0pt}{50pt}{40pt}
