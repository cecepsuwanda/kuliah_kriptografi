from Crypto.PublicKey import RSA

from Crypto.Cipher import PKCS1\_OAEP

from Crypto.Util.number import getPrime, inverse

def PembangkitanKunciRSA():

    key\_size = 1024  \# Gunakan 1024-bit untuk menghindari error

    p = getPrime(key\_size // 2)

    q = getPrime(key\_size // 2)

    n = p * q

    phi\_n = (p - 1) * (q - 1)

    e = 65537  \# Kunci publik, default, bisa diganti

    d = inverse(e, phi\_n)  \# Kunci privat

    \# Buat kunci RSA

    key = RSA.construct((n, e, d, p, q))
