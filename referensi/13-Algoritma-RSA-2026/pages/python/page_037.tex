def encrypt\_message(message, file\_kunci):

    \#Enkripsi pesan

    with open(file\_kunci, "rb") as file:

         kunci\_publik = RSA.importKey(file.read())


    RSA\_cipher = PKCS1\_OAEP.new(kunci\_publik)

    ciphertext = RSA\_cipher.encrypt(message.encode())

    return ciphertext 



def decrypt\_message(ciphertext, file\_kunci):

    \#Dekripsi pesan

    with open(file\_kunci, "rb") as file:

         kunci\_privat = RSA.importKey(file.read(), 'passwordku')


    RSA\_cipher = PKCS1\_OAEP.new(kunci\_privat)

    plaintext = RSA\_cipher.decrypt(ciphertext)

    return plaintext
