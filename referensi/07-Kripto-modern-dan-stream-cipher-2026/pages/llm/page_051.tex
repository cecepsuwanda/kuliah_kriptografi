i = 0; j = 0;
while (!feof(Fin)) {
  p = fgetc(Fin);
  i = (i + 1) % 256;
  j = (j + S[i]) % 256;
  swap(S[i], S[j]); // Pertukarkan nilai S[i] dan S[j]
  t = (S[i] + S[j]) % 256;
  u = S[t]; // keystream
  c = u ^ p;
  fputc(c, Fout);
}
fclose(Fin); fclose(Fout);
