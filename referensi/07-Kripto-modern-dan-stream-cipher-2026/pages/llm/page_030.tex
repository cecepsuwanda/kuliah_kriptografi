\begin{itemize}
    \item Misalkan plainteks adalah:
    \[ P: 0101\ 0111\ 0001\ 1001\ 0110\ 0100...\ (571964\text{ dalam Hexa}) \]
    Keystream yang dibangkitkan:
    \[ K: 111101011001001000111101011... \]
    \item Enkripsi: $C = P \oplus K$
    \vspace{10mm}
    \item Dekripsi: $P = C \oplus K$
    \vspace{10mm}
\end{itemize}
