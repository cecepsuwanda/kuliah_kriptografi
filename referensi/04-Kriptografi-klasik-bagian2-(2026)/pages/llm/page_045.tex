\section*{Extended Vigenere Cipher}

\begin{itemize}
    \item Vigenere cipher untuk 26 huruf dapat dikembangkan untuk enkripsi 256 karakter ASCII
\end{itemize}

\vspace{10mm}

Enkripsi: \[ c_j = E(p_j) = (p_j + k_i) \pmod{256} \]

Dekripsi: \[ p_j = D(c_j) = (c_j - k_i) \pmod{256} \]
