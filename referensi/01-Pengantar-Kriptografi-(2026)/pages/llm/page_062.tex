\section*{3. Fungsi Hash}

\begin{itemize}
    \item Mengkompresi pesan ukuran sembarang menjadi \textit{message-digest} berukuran \textit{fixed}.
    \item \textit{Irreversible} (tidak bisa dikembalikan menjadi pesan semula)
\end{itemize}

\vspace{10mm}

\begin{itemize}
    \item MD5
    \item SHA-1
    \item SHA-2
    \item Keccak (SHA-3)
    \item RIPEMD
    \item WHIRLPOOL
\end{itemize}
