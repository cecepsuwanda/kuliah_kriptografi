\section*{Double DES}

\vspace{89.10mm}

\begin{itemize}
    \item Menggunakan 2 buah kunci eksternal, $K_1$ dan $K_2$.
    \item Enkripsi: \[ C = E_{K_2}(E_{K_1}(P)) \]
    \item Dekripsi: \[ P = D_{K_1}(D_{K_2}(C)) \]
\end{itemize}
