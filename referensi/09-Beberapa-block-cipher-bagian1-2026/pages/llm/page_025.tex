\begin{itemize}
    \item $P(B)$ merupakan luaran dari fungsi $f$.
    \item Bit-bit $P(B)$ di-XOR-kan dengan $L_{i-1}$ menghasilkan $R_i$:
    \[ R_i = L_{i-1} \oplus P(B) \]
    \item Jadi, luaran dari putaran ke-$i$ adalah
    \[ (L_i, R_i) = (R_{i-1}, L_{i-1} \oplus P(B)) \]
\end{itemize}
