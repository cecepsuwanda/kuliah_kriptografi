def EkstraksiPesan(stego):    

  \# Baca citra stego, ekstrak pixel-pixel sebagai larik (array)    

  citra = Image.open(stego, 'r’) 

  citra = citra.convert("RGB")

  n = 3  

  citra.show()   

  larik\_pixel = np.array(list(citra.getdata()))   


  total\_pixel = larik\_pixel.size//n     \#jumlah pixel di dalam citra    

  \#Ekstraksi bit-bit pesan dari bit LSB setiap byte pixel   

  bit\_pesan = ""   

  for p in range(total\_pixel):       

     for q in range(0, 3):

        bit\_pesan = bit\_pesan + str(larik\_pixel[p][q] \& 1) \#ambil bit LSB        


  \# Kelompokkan setiap 8-bit dari bit\_pesan, simpan sebagai list          

  bit\_pesan = [bit\_pesan[i:i+8] for i in range(0, len(bit\_pesan), 8)]
