m = (yp – yq)/(xp – xq) 

    =(-1.86-0.836)/(-2.35-(-0.1))

    = -2.696 / -2.25 = 1.198

xr = m2 – xp – xq  

     = (1.198)2 – (-2.35) – (-0.1)

    = 3.89 

yr = m(xp – xr) – yp 

    = 1.198(-2.35 – 3.89) – (-1.86)

    = –5.62
