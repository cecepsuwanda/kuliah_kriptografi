\begin{align*}
 m &= (y_p - y_q)/(x_p - x_q) \\
 &= (-1.86 - 0.836)/(-2.35 - (-0.1)) \\
 &= -2.696 / -2.25 = 1.198 
\end{align*}

\begin{align*}
 x_r &= m^2 - x_p - x_q \\
 &= (1.198)^2 - (-2.35) - (-0.1) \\
 &= 3.89 
\end{align*}

\begin{align*}
 y_r &= m(x_p - x_r) - y_p \\
 &= 1.198(-2.35 - 3.89) - (-1.86) \\
 &= -5.62 
\end{align*}
